def sh(blob):return hashlib.sha256(blob).hexdigest()
def canonical(value):return json.dumps(value,sort_keys=True,separators=(',',':'),ensure_ascii=False,allow_nan=False).encode()
def same(actual,expected,code):check(canonical(actual)==canonical(expected),code)
def read(path,limit=8*1024*1024):
    path=Path(path)
    before=path.lstat();check(stat.S_ISREG(before.st_mode)and not path.is_symlink()and before.st_nlink==1 and 0<=before.st_size<=limit,'CLOSED_REGULAR_BOUNDED_FILE')
    keys=('st_mode','st_dev','st_ino','st_nlink','st_size','st_mtime_ns')
    key=lambda x:tuple(getattr(x,k)for k in keys)
    with path.open('rb')as handle:
        opened=os.fstat(handle.fileno());check(key(opened)==key(before),'READ_OPEN_IDENTITY')
        blob=handle.read(limit+1);after=os.fstat(handle.fileno())
        check(key(after)==key(opened)and after.st_ctime_ns==opened.st_ctime_ns,'READ_HANDLE_STABLE')
    final=path.lstat();check(key(final)==key(before)and final.st_ctime_ns==before.st_ctime_ns and len(blob)==before.st_size,'READ_CLOSED_IDENTITY')
    with path.open('rb')as handle:fresh=handle.read(limit+1)
    check(fresh==blob,'FRESH_CLOSED_BYTES')
    return blob

def document(blob):
    def pairs(items):
        value={}
        for key,item in items:
            check(key not in value,'DUPLICATE_JSON_KEY');value[key]=item
        return value
    value=json.loads(blob,object_pairs_hook=pairs,parse_constant=lambda _:check(False,'NONFINITE_JSON'))
    def typed(item,depth=0):
        check(depth<=64,'JSON_DEPTH')
        if type(item)is dict:
            for v in item.values():typed(v,depth+1)
        elif type(item)is list:
            for v in item:typed(v,depth+1)
        else:check(item is None or type(item)in(str,int,bool)or type(item)is float and math.isfinite(item),'JSON_FINITE_TYPES')
    typed(value);return value
