def _required_safety_evidence_complete():

    return persist_best_effort('REQUIRED_SAFETY_RECORD') if _SAFETY_IO_PROFILE=='OWNER' else best_effort_save()
def _required_safety_event(stage,status,**safe):

def _required_safety_failure(stage,code,primary=False):

        remember_failure(stage,code);return persist_best_effort('REQUIRED_SAFETY_FAILURE_RECORD')
    return record_failure(stage,code,primary=primary)

    if isinstance(error,Stop):_required_safety_failure(error.stage,error.code,primary=True)

class _RequiredSafetyBoundary:
    """Run the fixed safety action and preserve an already-pending exception."""
    def __init__(self,action,target,stage=None):
        need(action in ('CLOSE','FINALIZE'),'REQUIRED_SAFETY_BOUNDARY_ACTION');self.action=action;self.target=target;self.stage=stage
    def __enter__(self):return self
    def __exit__(self,kind,original,traceback):
        if original is not None:_required_safety_first_exception(original)
        try:
            if self.action=='CLOSE':closed_backends(self.target,self.stage)
            else:finalize(self.target)
        except BaseException:
            _required_safety_failure('REQUIRED_SAFETY_COMPOSITION','REQUIRED_SAFETY_ACTION_UNCONFIRMED')
            if original is None:raise
        return False


def finalize(name):
    if name in finalization_attempted:return
    finalization_attempted.add(name)
    active=_required_safety_sql(name,'FINALIZATION_BACKENDS')
    valid_active=(type(active) is dict and set(active)=={'migrator','other_client_backends'} and all(type(active[key]) is int and active[key]>=0 for key in ('migrator','other_client_backends')))
    if valid_active:report['targets'][name]['connections_before_finalization']=active;_required_safety_save()
    need(valid_active and all(active[key]==0 for key in ('migrator','other_client_backends')),'FINALIZATION_BLOCKED_ACTIVE_OPERATION',{'target':name})
    first=None;observation=None
    try:_required_safety_sql(name,'ORIGINAL_GRANTOR_FINALIZATION')
    except BaseException as error:first=error;_required_safety_first_exception(error)
    finally:
        try:observation=_required_safety_sql(name,'FINALIZATION_RESULT')
        except BaseException:
            _required_safety_failure('FINALIZATION_POST_OBSERVER','FINALIZATION_POST_OBSERVER_UNCONFIRMED')
            if first is None:raise
    if first is not None:raise first
    valid_observation=(type(observation) is dict and set(observation)=={'grantor','auth','remaining_memberships','leaf_create','active_install_connections'} and observation['grantor']=='postgres' and observation['auth']=='scram-sha-256:postgres' and type(observation['leaf_create']) is dict and set(observation['leaf_create'])==set(LEAVES) and all(type(value) is bool for value in observation['leaf_create'].values()) and all(type(observation[key]) is int and observation[key]>=0 for key in ('remaining_memberships','active_install_connections')))
    if valid_observation:report['targets'][name]['finalization_observation']=observation;_required_safety_save()
    need(valid_observation and observation['grantor']=='postgres' and observation['auth']=='scram-sha-256:postgres' and observation['remaining_memberships']==0 and observation['active_install_connections']==0 and not any(observation['leaf_create'].values()),'FINALIZATION_VERIFICATION_FAILED',{'target':name})
    finalized.add(name);report['targets'][name]['finalization']='PASS';_required_safety_event('FINALIZATION','PASS',target=name)
    if not _required_safety_evidence_complete():report['targets'][name]['finalization']='OBSERVED_CLEANUP_EVIDENCE_INCOMPLETE';_required_safety_save()
    _required_safety_barrier()
