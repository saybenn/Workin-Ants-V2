class Stop(Exception):
    def __init__(self,code,details=None):self.code=code;self.details=details or {};self.stage=report.get('stage','ENTRY');super().__init__(code)

def need(ok,code,details=None):
    if not ok:raise Stop(code,details)

def save():
    temp=OUT/'run-evidence.json.tmp'
    with temp.open('w',encoding='utf8',newline='\n') as f:
        text=json.dumps(report,indent=2)+'\n'
        for secret in all_secrets:text=text.replace(secret,'[REDACTED]')
        f.write(text);f.flush();os.fsync(f.fileno())
    temp.replace(OUT/'run-evidence.json')

def event(stage,status,**safe):
    report['stage']=stage;report['events'].append({'stage':stage,'status':status,**safe});save()
    print(json.dumps({'stage':stage,'status':status,**safe}),flush=True)

def command(args,input=None,timeout=60):
    return subprocess.run(args,input=input,capture_output=True,text=True,encoding='utf8',errors='strict',timeout=timeout,cwd=ROOT)

def docker(args,input=None,timeout=60):return command(['docker','--context','desktop-linux',*args],input,timeout)

def json_ok(result,code):
    need(result.returncode==0,code,{'exit_code':result.returncode})
    return json.loads(result.stdout)

def inventory():
    containers=docker(['container','ls','-a','--format','{{.Names}}'])
    volumes=docker(['volume','ls','--format','{{.Name}}'])
    need(containers.returncode==0 and volumes.returncode==0,'RESOURCE_INVENTORY_FAILED')
    return set(containers.stdout.splitlines()),set(volumes.stdout.splitlines())

def owned(value):return value.get('workinants.rehearsal')==LABEL and value.get('workinants.rehearsal.run')==RUN_ID

def inspect_container(name):
    need(name in RESOURCE_CONTAINERS,'UNDECLARED_CONTAINER_INSPECTION')
    result=docker(['container','inspect',name])
    if type(result.returncode) is int and result.returncode==0:
        value=json.loads(result.stdout);need(type(value) is list and len(value)==1 and type(value[0]) is dict and value[0].get('Name')=='/'+name,'CONTAINER_INSPECTION_IDENTITY');return value[0]
    containers,volumes=inventory();need(name not in containers,'CONTAINER_METADATA_UNCONFIRMED');return None

def inspect_volume(name):
    need(name in RESOURCE_VOLUMES,'UNDECLARED_VOLUME_INSPECTION')
    result=docker(['volume','inspect',name])
    if type(result.returncode) is int and result.returncode==0:
        value=json.loads(result.stdout);need(type(value) is list and len(value)==1 and type(value[0]) is dict and value[0].get('Name')==name,'VOLUME_INSPECTION_IDENTITY');return value[0]
    containers,volumes=inventory();need(name not in volumes,'VOLUME_METADATA_UNCONFIRMED');return None

def safe_error(stderr):
    found=[]
    for line in stderr.splitlines():
        m=re.fullmatch(r'(?:psql:(?:<stdin>|-):([0-9]+): )?(?:ERROR|FATAL):\s+([0-9A-Z]{5}):\s+(.*)',line)
        if not m:continue
        state=m[2];message=m[3];diagnostic='UNLISTED';obj=None
        known={'division by zero':'DIVISION_BY_ZERO','foreign Customer principal':'CUSTOMER_PRINCIPAL_GUARD','missing durable default-Free intent':'CUSTOMER_DEFAULT_FREE_GUARD','cleanup requires original bootstrap grantor':'WRONG_FINALIZATION_GRANTOR','temporary installation membership remains':'TEMPORARY_MEMBERSHIP_REMAINS','temporary function creation authority remains':'TEMPORARY_CREATE_REMAINS'}
        if message in known:diagnostic=known[message]
        found.append({'sqlstate':state,'diagnostic':diagnostic,'source_line':int(m[1]) if m[1] else None,**({'object':obj} if obj else {})})
    return found

def psql_args(target,role):
    return ['docker','--context','desktop-linux','exec','-i','-e','PGPASSFILE=/run/wa-secrets/pgpass','-e','PGSSLMODE=disable','-e','PGGSSENCMODE=disable','-e','PGAPPNAME='+CONNECTION_TAG,created_containers[target],'psql','-X','-w','-qAt','-v','ON_ERROR_STOP=1','-v','VERBOSITY=verbose','-v','SHOW_CONTEXT=never','-h','127.0.0.1','-p','5432','-U',role,'-d',DB]

def sql(target,role,text,kind,timeout=120,expected_error=None,installation=False):
    report['stage']=kind;save()
    args=psql_args(target,role)
    if installation:
        args+=['-c',"SELECT jsonb_build_object('session_user',session_user,'current_user',current_user,'system_user',system_user,'database',current_database());",'-f','-']
    result=command(args,text,timeout)
    details={'target':target,'role':role,'operation':kind,'exit_code':result.returncode,'errors':safe_error(result.stderr),'client_exited':True}
    if kind!='CREDENTIAL_PROVISION':details['input_sha256']=hashlib.sha256(text.encode()).hexdigest()
    if installation and result.stdout.splitlines():
        try:
            identity=json.loads(result.stdout.splitlines()[0])
            if set(identity)=={'session_user','current_user','system_user','database'}:details['actual_install_connection']=identity
        except json.JSONDecodeError:pass
    report['events'].append(details);save()
    if expected_error:
        need(result.returncode==3 and len(details['errors'])==1 and details['errors'][0]['sqlstate']==expected_error and details['errors'][0]['diagnostic']=='DIVISION_BY_ZERO','EXPECTED_INSTALL_FAILURE_NOT_PROVEN',details)
    else:need(result.returncode==0,'SQL_OPERATION_FAILED',details)
    return result.stdout.strip(),details

def observe(target,text,kind):
    stdout,details=sql(target,'postgres',text,kind)
    return json.loads(stdout)

def secure_private_directory():
    PRIVATE.mkdir()
    ps="""$ErrorActionPreference='Stop'
$privateTarget=[System.IO.Path]::GetFullPath($args[0])
$sid=[System.Security.Principal.WindowsIdentity]::GetCurrent().User
$acl=[System.Security.AccessControl.DirectorySecurity]::new()
$acl.SetOwner($sid)
$acl.SetAccessRuleProtection($true,$false)
$rule=[System.Security.AccessControl.FileSystemAccessRule]::new($sid,'FullControl','ContainerInherit,ObjectInherit','None','Allow')
$acl.AddAccessRule($rule)
[System.IO.Directory]::SetAccessControl($privateTarget,$acl)
$actual=[System.IO.Directory]::GetAccessControl($privateTarget)
$rules=$actual.GetAccessRules($true,$true,[System.Security.Principal.SecurityIdentifier])
if ($rules.Count -ne 1 -or $rules[0].IdentityReference.Value -ne $sid.Value -or -not $actual.AreAccessRulesProtected -or $actual.GetOwner([System.Security.Principal.SecurityIdentifier]).Value -ne $sid.Value) { throw 'Owner-only ACL verification failed' }
Write-Output 'OWNER_ONLY_VERIFIED'
"""
    # No secret is passed through this command. The only argument is a fixed
    # test-owned directory; the inherited DACL protects later-created files.
    helper=OUT/'set-private-acl.ps1';helper.write_text(ps,encoding='utf8',newline='\n')
    result=command(['powershell.exe','-NoProfile','-NonInteractive','-File',str(helper),str(PRIVATE)])
    need(result.returncode==0 and result.stdout.strip()=='OWNER_ONLY_VERIFIED','OWNER_ONLY_ACL_FAILED')
    report['credential_handling']={'host_private_directory_owner_only_acl_verified':True,'real_credentials_in_argv_env_or_evidence':False,'container_pgpass':'root_owned_0600_tmpfs','independent_per_role_per_target':True};save()

def preflight():
    context=json_ok(docker(['context','inspect','desktop-linux']),'CONTEXT_FAILED')[0]
    need(context['Endpoints']['docker']['Host']=='npipe:////./pipe/dockerDesktopLinuxEngine','NONLOCAL_DOCKER_ENDPOINT')
    server=json_ok(docker(['version','--format','{{json .Server}}']),'DOCKER_SERVER_FAILED')
    need(server['Os']=='linux' and server['Arch']=='amd64','DOCKER_SERVER_PLATFORM_MISMATCH')
    containers,volumes=inventory()
    presence={name:name in containers for name in RESOURCE_CONTAINERS}|{name:name in volumes for name in RESOURCE_VOLUMES}
    report['local_preflight']={'endpoint':context['Endpoints']['docker']['Host'],'server_version':server['Version'],'server_os':server['Os'],'server_arch':server['Arch'],'initial_resource_presence':presence,'unrelated_containers_untouched':len(containers),'unrelated_volumes_untouched':len(volumes)}
    need(not any(presence.values()),'RESOURCE_NAME_CONFLICT',{'presence':presence})
    event('LOCAL_PREFLIGHT','PASS',all_created_names_absent=True,expected_resource_names=len(RESOURCE_CONTAINERS)+len(RESOURCE_VOLUMES))

def ensure_image():
    # This dormant operator requires the exact already-cached approved image.
    # No registry requests, pull fallback or shared-image changes.
    values=json_ok(docker(['image','inspect',IMAGE]),'PINNED_CACHED_IMAGE_REQUIRED')
    need(len(values)==1,'PINNED_IMAGE_SINGLE_IDENTITY')
    value=values[0]
    need(value['Os']=='linux' and value['Architecture']=='amd64' and value['Id']==CONFIG,'PINNED_IMAGE_CONFIG_PLATFORM_MISMATCH')
    need(any(item.endswith('@'+IMAGE.split('@')[1]) for item in value['RepoDigests']),'PINNED_IMAGE_DIGEST_MISSING')
    report['image']={'reference':IMAGE,'image_id':value['Id'],'os':'linux','architecture':'amd64','pinned_repo_digest_and_config_verified':True,'pulled_this_run':False,'removed':False};event('PINNED_IMAGE','PASS',pulled=False,image_id=value['Id'])

    for target in TARGETS:
        name=target['name'];volume=target['volume'];passwords[name]={role:secrets.token_hex(32) for role in ROLES};all_secrets.extend(passwords[name].values())

        with secret.open('x',encoding='utf8',newline='\n') as f:f.write(passwords[name]['postgres']+'\n')
        # Assert absence immediately before each creation as well as globally.
        containers,volumes=inventory();need(name not in containers and volume not in volumes,'LATE_RESOURCE_NAME_CONFLICT')
        attempted_volumes.add(volume)
        r=docker(['volume','create','--label','workinants.rehearsal='+LABEL,'--label','workinants.rehearsal.run='+RUN_ID,volume])

        args=['container','create','--name',name,'--label','workinants.rehearsal='+LABEL,'--label','workinants.rehearsal.run='+RUN_ID,'--network','none','--platform','linux/amd64','--mount','type=volume,src='+volume+',dst=/var/lib/postgresql/data','--mount','type=bind,src='+str(secret)+',dst=/run/bootstrap-password,readonly','--tmpfs','/run/wa-secrets:rw,nosuid,noexec,nodev,mode=0700','-e','POSTGRES_DB='+DB,'-e','POSTGRES_PASSWORD_FILE=/run/bootstrap-password','-e','POSTGRES_HOST_AUTH_METHOD=scram-sha-256','-e','POSTGRES_INITDB_ARGS=--auth-host=scram-sha-256',IMAGE,'postgres','-c','password_encryption=scram-sha-256','-c','log_statement=none','-c','log_min_error_statement=panic','-c','log_error_verbosity=terse','-c','log_line_prefix=WA_F0_DIAG|%e|%u|%d|','-c','lc_messages=C','-c','log_min_messages=warning']

        value=inspect_container(name);need(value is not None and owned(value['Config']['Labels']),'CONTAINER_OWNERSHIP_UNCONFIRMED');created_containers[name]=value['Id']
        need(value['Config']['Image']==IMAGE and value['Image']==report['image']['image_id'] and value['HostConfig']['NetworkMode']=='none' and not value['HostConfig'].get('PortBindings'),'CONTAINER_ISOLATION_MISMATCH')

        need(docker(['container','start',created_containers[name]]).returncode==0,'CONTAINER_START_FAILED')

def finalize(name):
    if name in finalization_attempted:return
    finalization_attempted.add(name)
    active=observe(name,"SELECT jsonb_build_object('migrator',(SELECT count(*) FROM pg_stat_activity WHERE datname=current_database() AND usename='wa_migrator'),'other_client_backends',(SELECT count(*) FROM pg_stat_activity WHERE backend_type='client backend' AND pid<>pg_backend_pid()));",'PRE_FINALIZATION_CONNECTION_OBSERVER')
    valid_active=(type(active) is dict and set(active)=={'migrator','other_client_backends'} and all(type(active[key]) is int and active[key]>=0 for key in ('migrator','other_client_backends')))
    if valid_active:report['targets'][name]['connections_before_finalization']=active;save()
    need(valid_active and all(active[key]==0 for key in ('migrator','other_client_backends')),'FINALIZATION_BLOCKED_ACTIVE_OPERATION',{'target':name})
    sql(name,'postgres',read_pinned('review-output/foundation-bootstrap-finalize.review.sql').decode('utf8'),'ORIGINAL_GRANTOR_FINALIZATION')
    observation=observe(name,"SELECT jsonb_build_object('grantor',session_user,'auth',system_user,'remaining_memberships',(SELECT count(*) FROM pg_auth_members WHERE member='wa_schema_owner'::regrole AND roleid IN ('wa_private_helper_owner'::regrole,'wa_customer_verifier'::regrole,'wa_track_verifier'::regrole)),'leaf_create',jsonb_build_object('wa_private_helper_owner',has_schema_privilege('wa_private_helper_owner','platform_private','CREATE'),'wa_customer_verifier',has_schema_privilege('wa_customer_verifier','platform_private','CREATE'),'wa_track_verifier',has_schema_privilege('wa_track_verifier','platform_private','CREATE')),'active_install_connections',(SELECT count(*) FROM pg_stat_activity WHERE datname=current_database() AND usename='wa_migrator'));",'FINALIZATION_OBSERVER')
    valid_observation=(type(observation) is dict and set(observation)=={'grantor','auth','remaining_memberships','leaf_create','active_install_connections'} and observation['grantor']=='postgres' and observation['auth']=='scram-sha-256:postgres' and type(observation['leaf_create']) is dict and set(observation['leaf_create'])==set(LEAVES) and all(type(value) is bool for value in observation['leaf_create'].values()) and all(type(observation[key]) is int and observation[key]>=0 for key in ('remaining_memberships','active_install_connections')))
    if valid_observation:report['targets'][name]['finalization_observation']=observation;save()
    need(valid_observation and observation['grantor']=='postgres' and observation['auth']=='scram-sha-256:postgres' and observation['remaining_memberships']==0 and observation['active_install_connections']==0 and not any(observation['leaf_create'].values()),'FINALIZATION_VERIFICATION_FAILED',{'target':name})
    finalized.add(name);report['targets'][name]['finalization']='PASS';event('FINALIZATION','PASS',target=name)

def secret_scan():
    scanned=0
    for path in OUT.rglob('*'):
        if not path.is_file() or PRIVATE in path.parents:continue
        text=path.read_text(encoding='utf8');scanned+=1
        for value in all_secrets:
            need(value not in text,'SECRET_EVIDENCE_SCAN_FAILED',{'file':str(path.relative_to(OUT))})
    report['sanitization']={'files_scanned':scanned,'actual_password_values_found':False,'unrestricted_subprocess_output_persisted':False,'raw_context_handles_persisted':False,'coordinator_evidence_uses_reviewed_allowlist':True};save()

def cleanup():
    errors=[]
    def attempt(label,target,action):
        try:action()
        except BaseException as exc:
            item={'stage':'CLEANUP','target':target,'code':exc.code if isinstance(exc,Stop) else label}
            errors.append(item);report['cleanup'][target]={'status':'BLOCKED','failure':item['code']};save()
    def container_cleanup(name):
        value=inspect_container(name)
        if name not in created_containers:
            if value is None or not owned(value['Config']['Labels']):return
            created_containers[name]=value['Id']
        if value is None:
            report['cleanup'][name]={'status':'ABSENT_AT_CLEANUP'};save();return
        identity=created_containers[name]
        need(value.get('Name')=='/'+name and value['Id']==identity and owned(value['Config']['Labels']),'CLEANUP_CONTAINER_OWNERSHIP_MISMATCH',{'target':name})
        need(docker(['container','rm','-f',identity]).returncode==0,'EXACT_CONTAINER_REMOVAL_FAILED',{'target':name})
        report['cleanup'][name]={'removed':True,'verified_id':identity,'name_and_both_labels_verified':True};save()
    def volume_cleanup(name):
        value=inspect_volume(name)
        if name not in created_volumes:
            if value is None or not owned(value.get('Labels') or {}):return
            created_volumes[name]=value['CreatedAt']
        if value is None:
            report['cleanup'][name]={'status':'ABSENT_AT_CLEANUP'};save();return
        need(value.get('Name')==name and value['CreatedAt']==created_volumes[name] and owned(value.get('Labels') or {}),'CLEANUP_VOLUME_OWNERSHIP_MISMATCH',{'target':name})
        need(docker(['volume','rm',name]).returncode==0,'EXACT_VOLUME_REMOVAL_FAILED',{'target':name})
        report['cleanup'][name]={'removed':True,'name_labels_creation_identity_verified':True};save()
    try:
        for name in sorted(attempted_containers):attempt('CONTAINER_CLEANUP_FAILED',name,lambda name=name:container_cleanup(name))
        for name in sorted(attempted_volumes):attempt('VOLUME_CLEANUP_FAILED',name,lambda name=name:volume_cleanup(name))
    finally:
        deleted=[]
        for path in secret_paths:
            def delete_secret(path=path):
                private_resolved=PRIVATE.resolve()
                need(private_resolved==OUT.resolve()/'private' and private_resolved.is_relative_to(TASK.resolve()) and not private_resolved.is_relative_to(ROOT.resolve()),'SECRET_CLEANUP_DIRECTORY_MISMATCH')
                need(path.resolve().parent==private_resolved and path.name in ('ok-bootstrap-password',),'SECRET_CLEANUP_PATH_MISMATCH')
                if path.exists():path.unlink()
                deleted.append(path.name)
            attempt('SECRET_FILE_REMOVAL_FAILED',path.name,delete_secret)
        if PRIVATE.exists():attempt('PRIVATE_DIRECTORY_REMOVAL_FAILED','private_directory',PRIVATE.rmdir)
        report['host_secret_cleanup']={'deleted':deleted,'private_directory_absent':not PRIVATE.exists()};save()
    try:
        containers,volumes=inventory()
        presence={name:name in containers for name in RESOURCE_CONTAINERS}|{name:name in volumes for name in RESOURCE_VOLUMES}
        report['cleanup_summary']={'resource_presence_after':presence,'test_owned_secret_files_deleted':report['host_secret_cleanup']['deleted'],'private_directory_absent':not PRIVATE.exists(),'image_removed':False,'unrelated_resources_changed':False}
        need(not any(presence.values()),'RESOURCE_ABSENCE_NOT_CONFIRMED')
    except BaseException as exc:
        errors.append({'stage':'CLEANUP','code':exc.code if isinstance(exc,Stop) else 'CLEANUP_ABSENCE_OBSERVATION_FAILED'})
    if errors:
        report['unexpected'].extend(errors);save();raise Stop('CONTROLLED_CLEANUP_INCOMPLETE')
    report['steps']['CATALOG-CLEANUP']='PASS';event('CONTROLLED_CLEANUP','PASS',all_created_names_absent=True,expected_resource_names=len(RESOURCE_CONTAINERS)+len(RESOURCE_VOLUMES),test_secret_files_absent=True)

def launch(argv=None):
    parser=argparse.ArgumentParser(description='Dormant finite lease/revocation operator. Source review and exact literal approval required.')
    parser.add_argument('--execute-reviewed-lease-revoke',action='store_true')
    parser.add_argument('--manifest',type=Path)
    args=parser.parse_args(argv)
    if not args.execute_reviewed_lease_revoke or args.manifest is None:
        print(json.dumps({'status':'HELD','code':'EXPLICIT_REVIEWED_EXECUTION_REQUIRED'}));return 1
