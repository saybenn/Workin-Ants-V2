def unique(raw):

class Evidence:
    def __init__(self,writer):

    def save(self):
        try:self.writer(self.value)
        except BaseException:
            self.value['evidence_complete']=False
            if self.value['first_failure']is None:self.value['first_failure']={'stage':'EVIDENCE','code':'PERSISTENCE_UNCONFIRMED'}
            return False
        return self.value['evidence_complete']
    def record(self,record):self.value['operations'].append(record);return self.save()
