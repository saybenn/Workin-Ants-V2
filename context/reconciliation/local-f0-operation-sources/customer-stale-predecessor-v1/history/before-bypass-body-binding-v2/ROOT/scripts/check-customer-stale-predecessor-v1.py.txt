def constant(node):
    if isinstance(node,pgast.TypeCast):return constant(node.arg)
    if isinstance(node,pgast.A_ArrayExpr):return [constant(x)for x in node.elements]
    if isinstance(node,pgast.A_Const):
        if node.isnull:return None
        for key in('sval','ival','boolval'):
            if hasattr(node.val,key):return getattr(node.val,key)
    raise AssertionError('UNMODELED_GRANT_VALUE')

class Output:
    def __init__(self,world,error=False):self.q=queue.Queue();self.world=world;self.error=error;self.once=False
    def readline(self,n):return self.q.get()
    def read(self,n):
        self.world.ended.wait(2)
        if not self.world.ended.is_set():raise AssertionError('MOCK_STDERR_EOF_WAS_NOT_REQUESTED')
        if self.once:return ''
        self.once=True;return self.world.errors

class Input:
    def __init__(self,process):self.process=process;self.closed=False
    def write(self,text):
        if self.closed:raise ValueError('INPUT_CLOSED')
        self.process.execute(text)
    def flush(self):pass
    def close(self):self.closed=True
