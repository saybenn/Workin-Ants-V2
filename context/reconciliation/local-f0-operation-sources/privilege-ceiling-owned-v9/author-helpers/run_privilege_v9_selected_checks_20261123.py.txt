class Result(unittest.TestResult):
    def __init__(self): super().__init__(); self.records = []
    def addSuccess(self, test):
        super().addSuccess(test); self.records.append({'test': test.id(), 'status': 'PASS'})
    def failed(self, test, err, status):
        frames = traceback.extract_tb(err[2]); last = frames[-1] if frames else None
        code = getattr(err[1], 'code', None)
        self.records.append({'test': test.id(), 'status': status, 'class': err[0].__name__, 'code': code if type(code) is str and len(code) < 160 and all(c.isupper() or c.isdigit() or c == '_' for c in code) else None, 'frame': {'file': Path(last.filename).name, 'function': last.name, 'line': last.lineno} if last else None, 'traceback_sha256': H(''.join(traceback.format_exception(*err)).encode())})
    def addFailure(self, test, err):
        super().addFailure(test, err); self.failed(test, err, 'FAIL')
    def addError(self, test, err):
        super().addError(test, err); self.failed(test, err, 'ERROR')
