def need(value,code):

def audit(event,args):

def sha(raw):return hashlib.sha256(raw).hexdigest()
def encoded(value):return (json.dumps(value,indent=2,allow_nan=False)+'\n').encode()
def decoded(raw):
    def pairs(items):
        value={}
        for key,item in items:need(key not in value,'DUPLICATE_JSON_KEY');value[key]=item
        return value
    def constant(value):raise Refused('NONFINITE_JSON')
    return json.loads(raw,object_pairs_hook=pairs,parse_constant=constant)
def safe_relative(name):
    need(type(name)is str and name and not Path(name).is_absolute() and ':'not in name and '\\'not in name and all(p not in ('','.','..')for p in name.split('/')) and not any(ord(c)<32 for c in name),'SAFE_RELATIVE_PATH_REQUIRED')
    return name
def safe_target(base,name):
    safe_relative(name);target=base/name;resolved=target.resolve()
    need(resolved.is_relative_to(base.resolve()) and resolved!=base.resolve(),'TARGET_PATH_ESCAPE')
    for part in (target,*target.parents):
        if part==base.parent:break
        need(not part.is_symlink() and not part.is_junction(),'LINKED_WRITE_PATH_REFUSED')
    return target
def same_or_absent(path,raw):
    need(not path.exists() or(path.is_file() and path.read_bytes()==raw),'EXISTING_WORK_CONFLICT')
def write_new_or_identical(path,raw):
    same_or_absent(path,raw)
    if path.exists():return
    path.parent.mkdir(parents=True,exist_ok=True)
    with path.open('xb')as handle:handle.write(raw)
def replace_exact(path,before,after):
    need(path.read_bytes()==before,'METADATA_CHANGED_BEFORE_REPLACEMENT')
    temporary=path.with_name(path.name+'.v9-admission.tmp')
    need(not temporary.exists(),'METADATA_TEMP_CONFLICT')
    with temporary.open('xb')as handle:handle.write(after)
    temporary.replace(path)
