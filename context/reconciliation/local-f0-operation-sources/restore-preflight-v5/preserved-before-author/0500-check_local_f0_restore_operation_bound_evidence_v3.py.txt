def archive_bytes(hba,extra=None,omit=()):
    entries={'PG_VERSION':b'17\n','backup_manifest':b'{}','postgresql.auto.conf':b'# inert\n','pg_hba.conf':hba.encode(),
        'global/pg_control':b'\0'*8192,'pg_wal/'+'A'*24:b'WAL_FIXTURE_NOT_RUNTIME_PROOF','base/5/1':CANARY.encode()}
    entries.update(extra or {})
    out=io.BytesIO()
    with tarfile.open(fileobj=out,mode='w') as tf:
        for name,value in entries.items():
            if name in omit:continue
            info=tarfile.TarInfo(name);info.size=len(value);tf.addfile(info,io.BytesIO(value))
    return out.getvalue()
