class Refused(Exception): pass
def need(value,code):
    if not value:raise Refused(code)
def sha(raw):return hashlib.sha256(raw).hexdigest()
def data(value):return (json.dumps(value,indent=2,allow_nan=False)+'\n').encode()


def source_blob(relative,expected=None):
    need(type(relative) is str and not Path(relative).is_absolute() and chr(92) not in relative and all(part not in ('','.','..') for part in relative.split('/')),'RESTORE_SOURCE_PATH')
    path=(ROOT/relative).resolve();need(path.is_relative_to(ROOT.resolve()),'RESTORE_SOURCE_PATH')
    raw=path.read_bytes()
    if expected is not None:need(sha(raw)==expected,'RESTORE_SOURCE_ADMISSION_DRIFT')
    return raw



def preflight_v5_source_admission(read_source,approved_required,proposal,all_five):
    """Data-only, exact V5 admission. No project code executes in this gate."""
    prefix='context/reconciliation/local-f0-operation-sources/restore-preflight-v5/'
    child_path='context/reconciliation/local-f0-restore-preflight-v5-source-validation.json'

    sources={'scripts/prepare-restore-control-rehearsal-v5.py','scripts/check-restore-control-rehearsal-v5.py','scripts/sql-review/RESTORE-CONTROL-REHEARSAL-V5.md'}
    def decode(raw):
        def pairs(items):
            result={}
            for key,value in items:
                need(key not in result,'V5_DUPLICATE_METADATA_KEY');result[key]=value
            return result
        return json.loads(raw,object_pairs_hook=pairs,parse_constant=lambda _:need(False,'V5_NONFINITE_METADATA'))
    def pins(value):
        need(type(value) is dict and value and all(type(k) is str and type(v) is str and re.fullmatch('[0-9a-f]{64}',v) and not Path(k).is_absolute() and ':' not in k and chr(92) not in k and all(p not in ('','.','..') for p in k.split('/')) for k,v in value.items()),'V5_EXACT_PIN_MAP')
        return value
    child_raw=read_source(child_path);owned_raw=read_source(owned_path)
    child=decode(child_raw);owned=decode(owned_raw)
    need(type(approved_required) is bool and type(child) is dict and type(owned) is dict,'V5_METADATA_OBJECTS')
    need(child.get('schema')=='workinants.restore-preflight-v5-child-source-validation.v1' and owned.get('schema')=='workinants.restore-preflight-v5-owned-source-validation.v1','V5_METADATA_SCHEMAS')
    need(set(pins(child.get('sources_sha256')))==sources and child['sources_sha256']=={k:proposal['source_files_sha256'][k] for k in sources},'V5_CHILD_EXACT_THREE_SOURCES')
    need(child.get('packet_source_files_sha256')==proposal['source_files_sha256'] and child.get('prepared_files_sha256')==proposal['prepared_files_sha256'],'V5_CHILD_CLOSED_PACKET')
    need(type(all_five) is dict and len(all_five)==5 and owned.get('all_five_sources_sha256')==all_five,'V5_OWNED_EXACT_FIVE')
    archive=pins(owned.get('archive_files_sha256'))
    need(child.get('archive_files_sha256')==archive and all(k.startswith(prefix) for k in archive),'V5_EXACT_SHARED_ARCHIVE')
    disk=ROOT/prefix
    need(disk.is_dir() and not disk.is_symlink() and not any(p.is_symlink() for p in disk.rglob('*')),'V5_ARCHIVE_NO_SYMLINK')
    need({p.relative_to(ROOT).as_posix() for p in disk.rglob('*') if p.is_file()}==set(archive),'V5_ARCHIVE_EXACT_FILE_SET')
    result={child_path:hashlib.sha256(child_raw).hexdigest(),owned_path:hashlib.sha256(owned_raw).hexdigest()}
    for relative,expected in archive.items():
        need(hashlib.sha256(read_source(relative)).hexdigest()==expected,'V5_ARCHIVED_BYTE_DRIFT');result[relative]=expected
    for name,expected in all_five.items():
        need(type(name) is str and Path(name).name==name and type(expected) is str and re.fullmatch('[0-9a-f]{64}',expected),'V5_PHYSICAL_SOURCE_MAP')
        need(hashlib.sha256((TASK/name).read_bytes()).hexdigest()==expected and archive.get(prefix+name+'.txt')==expected,'V5_PHYSICAL_FIVE_SNAPSHOT_DRIFT')
    for name,expected in child['sources_sha256'].items():
        need(hashlib.sha256(read_source(name)).hexdigest()==expected and archive.get(prefix+'root-sources/'+name+'.txt')==expected,'V5_CHILD_SOURCE_SNAPSHOT_DRIFT')
    for meta in (child,owned):
        need(meta.get('source_review_status') in ('PENDING','SOURCE_APPROVED'),'V5_DECLARED_SOURCE_REVIEW_STATE')
        review=meta.get('independent_source_review')
        need(type(review) is dict and set(review)=={'decision','path','sha256'} and review['decision'] in ('PENDING','SOURCE_APPROVED'),'V5_INDEPENDENT_REVIEW_STATE')
        if approved_required:
            need(meta['source_review_status']=='SOURCE_APPROVED' and review['decision']=='SOURCE_APPROVED','V5_SEPARATE_SOURCE_REVIEW_REQUIRED')
            relative=review['path'];expected=review['sha256']
            need(type(relative) is str and relative.startswith(prefix+'reviews/') and type(expected) is str and re.fullmatch('[0-9a-f]{64}',expected) and archive.get(relative)==expected,'V5_REVIEW_ARCHIVE_BINDING')
            body=decode(read_source(relative))
            need(type(body) is dict and body.get('decision')=='SOURCE_APPROVED','V5_REVIEW_DECISION_REQUIRED')
        elif review['decision']=='PENDING':
            need(review['path'] is None and review['sha256'] is None,'V5_PENDING_REVIEW_HAS_NO_AUTHORITY')
    for name,expected in pins(proposal['source_files_sha256']).items():
        need(hashlib.sha256(read_source(name)).hexdigest()==expected,'V5_PACKET_SOURCE_DRIFT');result[name]=expected
    for name,expected in pins(proposal['prepared_files_sha256']).items():

    return result

    try:
        need(args.prepare_plan and args.baseline_manifest is not None and args.baseline_sha256 is not None,'EXPLICIT_REVIEWED_BASELINE_REQUIRED')
