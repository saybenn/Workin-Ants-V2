class Held(Exception):
    def __init__(self,code):self.code=code;super().__init__(code)
def need(value,code):
    if not value:raise Held(code)
def sha(raw):return hashlib.sha256(raw).hexdigest()
def encoded(value):return (json.dumps(value,indent=2,allow_nan=False)+'\n').encode()
def aggregate(files):return sha(''.join(path+'\0'+files[path]+'\n' for path in sorted(files)).encode())
def unique_json(raw):
    def pairs(items):
        value={}
        for key,item in items:need(key not in value,'DUPLICATE_JSON_KEY');value[key]=item
        return value
    return json.loads(raw,object_pairs_hook=pairs,parse_constant=lambda _:(_ for _ in ()).throw(Held('NONFINITE_JSON')))
def relative(value):

def source_path(value):

def external(value,parent=None):

def file_sha(path):
    digest=hashlib.sha256()
    with Path(path).open('rb') as handle:
        while block:=handle.read(1024*1024):digest.update(block)
    return digest.hexdigest()
def mapping(value,code):
    need(type(value) is dict and value and all(type(path) is str and type(digest) is str and SHA.fullmatch(digest) for path,digest in value.items()),code)
    for path in value:relative(path)
    return value
def git_read(*args):
    # rev-parse and blob reads only; no fetch, checkout, index or history writes.
    result=subprocess.run(['git','--no-optional-locks','-c','core.longpaths=true',*args],cwd=ROOT,capture_output=True,timeout=60,shell=False)
    need(type(result.returncode) is int and result.returncode==0,'READ_ONLY_GIT_FAILED');return result.stdout

def pin(path):
    path,raw,value,files=validate_filesystem(path)
    commit=git_read('rev-parse','HEAD').decode('ascii').strip();tree=git_read('rev-parse','HEAD^{tree}').decode('ascii').strip()
    need(re.fullmatch('[0-9a-f]{40}',commit) and re.fullmatch('[0-9a-f]{40}',tree),'COMMIT_AND_TREE_REQUIRED')
    for name,digest in files.items():need(file_sha(source_path(name))==digest==sha(git_read('show',commit+':'+name)),'CURRENT_AND_COMMITTED_BLOB_MISMATCH')
    need(git_read('rev-parse','HEAD').decode('ascii').strip()==commit,'HEAD_CHANGED_DURING_PIN')
    need(path.read_bytes()==raw,'HELD_LITERAL_CHANGED_DURING_PIN')
    # Recheck run-owned inputs after Git enumeration; neither source nor proof
    # can drift between initial filesystem admission and the final pin write.
    _,second_raw,_,second_files=validate_filesystem(path);need(second_raw==raw and second_files==files,'FILESYSTEM_CHANGED_DURING_PIN')
    before=path.with_name('operation-manifest-before-source-pin.json');temporary=path.with_name('operation-manifest-source-pin.tmp')
    need(not before.exists() and not temporary.exists(),'FRESH_PIN_PRESERVATION_REQUIRED')
    value.update(source_commit=commit,source_tree=tree,packet_files_sha256=dict(sorted(files.items())),packet_sha256=aggregate(files),pin_helper_sha256=sha(Path(__file__).read_bytes()))
    with before.open('xb') as handle:handle.write(raw)
    with temporary.open('xb') as handle:handle.write(encoded(value))
    temporary.replace(path)
    return {'status':'SOURCE_PINNED_PENDING_OPERATIONAL_REVIEW','source_commit':commit,'source_tree':tree,'packet_sha256':value['packet_sha256'],'packet_files':len(files),'manifest_sha256':file_sha(path),'review_status':value['review_status'],'runtime_approved':False,'database_contacted':False,'git_mutated':False}
def main(argv=None):
    parser=argparse.ArgumentParser(description=__doc__);parser.add_argument('manifest',type=Path);args=parser.parse_args(argv)
    try:print(json.dumps(pin(args.manifest)));return 0
    except BaseException as error:
        print(json.dumps({'status':'HELD','code':error.code if isinstance(error,Held) else 'RESTORE_SOURCE_PIN_REFUSED','raw_error_discarded':True,'runtime_approved':False}));return 1
if __name__=='__main__':raise SystemExit(main())
