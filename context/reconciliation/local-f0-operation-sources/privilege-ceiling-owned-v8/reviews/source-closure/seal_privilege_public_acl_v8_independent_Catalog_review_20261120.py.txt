def functions(path):
    result={}
    def walk(nodes,prefix=''):
        for node in nodes:
            if isinstance(node,(ast.FunctionDef,ast.AsyncFunctionDef,ast.ClassDef)):
                key=prefix+node.name;result[key]=ast.dump(node,include_attributes=False);walk(node.body,key+'.')
    walk(ast.parse(path.read_bytes()).body);return result
def normalized(value):
    return value.replace('v8','v7').replace('V8','V7').replace('b788cf40cda693be2c285c64572011ae54f7640bf3157000648196f603a7775c','571951d886af0d96eff0c5dad14783a8cdcefa49fe80dd7856fa104aed27c13f')
def delta(old,new):
    a=functions(old);b=functions(new)
    return {'added':sorted(set(b)-set(a)),'removed':sorted(set(a)-set(b)),'changed':sorted(k for k in a.keys()&b.keys()if a[k]!=normalized(b[k]))}
