def _operation_order(f):
    """Project append sites from pinned coordinator setup/execute/finally.

    This constructs a source recipe, not an expectation copied from output.
    Transactions append before issue/retention, and always append closure in
    finally. Observers each close other clients before reading state/catalog.
    """
    result=[];instances=0
    def add(stage,**kw):result.append({'stage':stage,**kw})
    def global_():add('GLOBAL_BACKEND_CLOSURE')
    def native(stage):global_();add(stage)
    def trigger():global_();add('ALL100_GUARDS_RESTORED')
    def retention(n,live=False):add('EXACT21_GRANT_RETENTION',instances=n,unexpired_required=live)
    def transaction(name,grant,control=None):
        nonlocal instances
        add('ACTUAL_TRANSACTION',case=name)
        if grant is not None:
            instances+=1;add('SEPARATE_ISSUER_COMMIT');retention(1,True)
        if control=='revoked_before_action':retention(1);add('SEPARATE_REVOKE_COMMIT_BEFORE_ACTION')
        if control=='server_expired':add('ACTUAL_SERVER_EXPIRY')
        global_()
    global_();native('PRE_SETUP_NATIVE');global_();trigger()
    transaction('DISABLE_EXACT15',None)
    for group in f['setup_groups']:
        global_();transaction(group['name'],group['grant_id']);global_();add('FIXTURE_GROUP_COMMITTED',name=group['name'])
    global_();add('EXACT15_RESTORE_ELIGIBILITY');transaction('RESTORE_EXACT15',None)
    trigger();native('RESTORED_FULL_NATIVE');add('EXACT48_SOURCE_COHERENCE')
    native('BEFORE_FIRST_SELECTED_READ');global_()
    for case in f['cases']:
        trigger();global_();transaction(case['name'],case['grant'],case['control'] if case['kind']=='binder-denial'else None)
        if case['kind']=='read'and case['grant'] is not None:retention(1,True)
        if instances:retention(instances)
        global_();add('EXACT48_SOURCE_COHERENCE');trigger()
    native('FINAL_FULL_NATIVE_PARITY');global_();trigger();native('FINALLY_FULL_NATIVE_IDENTITY')
    need(instances==79 and len(result)==1149,'SOURCE_ORDER_RECIPE_BOUNDS')
    return result

def _validate_order(value,f):
    order=_operation_order(f)
    same([{k:row[k]for k in item}for row,item in zip(value,order)],order,'EXACT1149_SOURCE_APPEND_ORDER')
    need(len(value)==len(order),'EXACT1149_OPERATION_COUNT')

class QualificationBatch:
    """Private immutable bytes; no handles/cache escape, sticky first refusal."""
    def __init__(self,root):
        path=Path(root);need(not path.is_symlink(),'ROOT_SYMLINK');self.__root=path.resolve(strict=True)
        b=_capture(self.__root);self.__execution=validate_bundle(b);e=doc(b[DERIVED])
        self.__rows=MappingProxyType({(c['case_id'],c['assertion_id']):(i,canonical(_expected_row(c,PINS[DERIVED],i)))for i,c in enumerate(e['claims'])})
        self.__bytes=MappingProxyType(b);self.__seen=set();self.__failed=None;self.__finalized=False
    def _reject(self,code):
        if self.__failed is None:self.__failed=code
        raise EvidenceFailure(self.__failed)from None
    def validate_claim(self,case_id,assertion_id,status,execution,root=None,json_pointer=None,ledger_row=None):
        try:
            need(self.__failed is None and not self.__finalized,'BATCH_CLOSED')
            need(all(type(x)is str for x in (case_id,assertion_id,status,json_pointer)),'CLAIM_TYPES');key=(case_id,assertion_id)
            need(key in self.__rows and key not in self.__seen,'EXACT_UNSEEN13_CLAIM');i,wanted=self.__rows[key]
            need(status=='PASS'and json_pointer=='/claims/'+str(i),'STATUS_AND_POINTER')
            if root is not None:need(Path(root).resolve(strict=True)==self.__root,'BATCH_ROOT')
            _typed_json(execution);_typed_json(ledger_row)
            need(type(execution)is dict and type(ledger_row)is dict and canonical(execution)==self.__execution and canonical(ledger_row)==wanted,'EXACT_DERIVED_AND_LEDGER_ROW')
            self.__seen.add(key);return True
        except EvidenceFailure as error:self._reject(str(error))
        except Exception:self._reject('READ1161_EVIDENCE_REJECTED')
    def verify_unchanged(self):
        try:
            need(self.__failed is None and not self.__finalized,'BATCH_CLOSED');need(self.__seen==set(self.__rows),'COMPLETE13_BATCH_REQUIRED')
            fresh=_capture(self.__root);need(all(fresh[p]==self.__bytes[p]for p in PINS),'FINAL_ALL_INPUT_BYTES_DRIFT')
            self.__bytes=None;self.__finalized=True;return True
        except EvidenceFailure as error:self._reject(str(error))
        except Exception:self._reject('READ1161_EVIDENCE_REJECTED')
    def require_finalized(self):
        if self.__failed is not None:raise EvidenceFailure(self.__failed)
        need(self.__finalized,'FINAL_FRESH_BYTE_BARRIER_REQUIRED');return True

def prepare_batch(root):
    try:return QualificationBatch(root)
    except EvidenceFailure:raise
    except Exception:raise EvidenceFailure('READ1161_EVIDENCE_REJECTED')from None

def validate_claim(case_id,assertion_id,status,execution,root,json_pointer=None,ledger_row=None):
    # Standalone fresh admission still admits all evidence and requires all13.
    # A caller cannot turn a single call into a completed qualification batch.
    b=prepare_batch(root);return b.validate_claim(case_id,assertion_id,status,execution,root,json_pointer,ledger_row)
