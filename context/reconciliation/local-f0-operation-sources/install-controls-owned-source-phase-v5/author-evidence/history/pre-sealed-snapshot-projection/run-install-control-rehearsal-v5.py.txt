class HeldFailure(Exception):
    def __init__(self,code):self.code=code;super().__init__(code)

def need(ok,code):
    if not ok:raise HeldFailure(code)

def sha(raw):return hashlib.sha256(raw).hexdigest()
def canonical(value):return json.dumps(value,sort_keys=True,separators=(',',':'),ensure_ascii=False,allow_nan=False)
def quote(value):return "'"+value.replace("'","''")+"'"
def exact(value,wanted,code):need(canonical(value)==canonical(wanted),code)

def unique_json(raw):
    def pairs(items):
        value={}
        for key,item in items:need(key not in value,'DUPLICATE_JSON_FIELD');value[key]=item
        return value
    return json.loads(raw,object_pairs_hook=pairs,parse_constant=lambda _:(_ for _ in ()).throw(HeldFailure('NONFINITE_JSON')))

def read_bound(binding,limit=16*1024*1024):
    need(type(binding) is dict and set(binding)=={'path','sha256'} and type(binding['sha256']) is str and re.fullmatch('[0-9a-f]{64}',binding['sha256']),'FILE_BINDING')
    path=external(binding['path']);need(not path.is_symlink() and path.is_file() and path.stat().st_size<=limit,'BOUNDED_REGULAR_FILE_REQUIRED')
    with path.open('rb') as f:raw=f.read(limit+1)
    need(len(raw)<=limit and sha(raw)==binding['sha256'],'FILE_HASH_DRIFT');return raw

def compiled(path,raw):
    module=ModuleType('_install_'+Path(path).stem.replace('-','_'));module.__file__=str(source(path));stream=io.StringIO()

    except BaseException:raise HeldFailure('PINNED_HELPER_LOAD_FAILED') from None
    need(not stream.getvalue(),'PINNED_HELPER_OUTPUT');return module.__dict__

class Evidence:
    def __init__(self,directory):
        self.required_cleanup=False;self.safe_codes=frozenset(('INSTALL_EVIDENCE_WRITE_FAILED',));self.safe_stages=frozenset(('ENTRY',));self.directory=directory;directory.mkdir(exist_ok=False);self.value={'schema':'workinants.install-controls-results.v2','evidence_complete':True,'persistence_failure':False,'first_failure':None,'status':'RUNNING','current_stage':'ENTRY','targets':{},'subsequent_cases_stopped':False,'limits':{'posture':'D','whole_cases':'NOT_PROMOTED','full_native_acl_and_ledger_comparison':'PENDING_OUTER_OPERATOR','451_register':'NOT_PROMOTED','raw_output_retained':False}};self.save()

    def save(self):
        try:
            path=self.directory/'install-controls-results.json';tmp=path.with_suffix('.tmp');raw=(json.dumps(self.value,indent=2,allow_nan=False)+'\n').encode('utf8')
            with tmp.open('wb')as handle:handle.write(raw);handle.flush();__import__('os').fsync(handle.fileno())
            tmp.replace(path);return raw
        except BaseException:
            self.value['evidence_complete']=False;self.value['persistence_failure']=True
            self.remember_failure(HeldFailure('INSTALL_EVIDENCE_WRITE_FAILED'))
            if self.required_cleanup:return b''
            raise HeldFailure('INSTALL_EVIDENCE_WRITE_FAILED')from None
    def remember_failure(self,error):
        code=getattr(error,'code',None);code=code if type(code)is str and code in self.safe_codes else 'INSTALL_PHASE_PROCESS_TIMEOUT'if type(error).__name__=='TimeoutExpired'else 'INSTALL_PHASE_UNEXPECTED_RESPONSE'
        stage=self.value.get('current_stage');stage=stage if type(stage)is str and stage in self.safe_stages else 'INSTALL_CHILD_OPERATION'
        if self.value.get('first_failure')is None:self.value['first_failure']={'stage':stage,'code':code,'raw_exception_discarded':True}
        self.value.update(status='STOPPED_UNEXPECTED_RESULT',subsequent_cases_stopped=True)
    def stage(self,name):self.value['current_stage']=name;return self.save()
