class EvidenceFailure(Exception):
    """Only fixed, secret-safe rejection codes escape the reader."""

def need(value, code):
    if not value: raise EvidenceFailure(code)

def sh(raw): return hashlib.sha256(raw).hexdigest()
def canonical(value): return json.dumps(value,sort_keys=True,separators=(',',':'),ensure_ascii=False,allow_nan=False)
def digest(value): return sh(canonical(value).encode('utf8'))
def encoded(value): return (json.dumps(value,indent=2,ensure_ascii=False,allow_nan=False)+'\n').encode('utf8')
def same(value, expected, code): need(canonical(value)==canonical(expected),code)
def integer(value, expected, code): need(type(value)is int and value==expected,code)
def keys(value, expected, code): need(type(value)is dict and set(value)==set(expected),'FIELDS_'+code)
def all_true(value, code): need(type(value)is dict and value and all(x is True for x in value.values()),code)
def hash_value(value, code): need(type(value)is str and re.fullmatch('[0-9a-f]{64}',value)is not None,code)

def typed(value, depth=0, budget=None):
    if budget is None: budget=[2000000]
    budget[0]-=1; need(depth<=64 and budget[0]>=0,'JSON_BOUNDS')
    kind=type(value)
    if kind is dict:
        need(all(type(k)is str and len(k)<=2048 for k in value),'JSON_KEYS')
        for x in value.values(): typed(x,depth+1,budget)
    elif kind is list:
        for x in value: typed(x,depth+1,budget)
    elif kind is str: need(len(value)<=1024*1024,'JSON_STRING_BOUNDS')
    elif kind is int: need(-(2**63)<=value<2**63,'JSON_INTEGER_BOUNDS')
    else: need(value is None or kind is bool or kind is float and math.isfinite(value),'JSON_TYPES')

def doc(raw):
    need(type(raw)is bytes and len(raw)<=8*1024*1024,'JSON_BYTES_LIMIT')
    def pairs(items):
        result={}
        for k,v in items:
            need(k not in result,'DUPLICATE_JSON'); result[k]=v
        return result
    try:
        result=json.loads(raw.decode('utf8'),object_pairs_hook=pairs,parse_constant=lambda _:(_ for _ in()).throw(EvidenceFailure('NONFINITE_JSON')))
        typed(result); return result
    except EvidenceFailure: raise
    except Exception: raise EvidenceFailure('MALFORMED_JSON') from None

def _read_bytes(root, relative, wanted):
    need(type(relative)is str and relative and not Path(relative).is_absolute() and ':'not in relative and '\\'not in relative and all(x not in ('','.','..') for x in relative.split('/')),'RELATIVE_PATH')
    hash_value(wanted,'EXPECTED_HASH'); path=root/relative
    def metadata():
        for p in (path,*path.parents):
            if not p.is_relative_to(root): break
            need(not p.is_symlink() and not getattr(p,'is_junction',lambda:False)(),'LINK_REJECTED')
        need(path.resolve(strict=True).is_relative_to(root),'PATH_ESCAPE'); return path.lstat()
    def identity(s):
        need(stat.S_ISREG(s.st_mode) and s.st_nlink==1 and 0<=s.st_size<=8*1024*1024,'REGULAR_SINGLE_LINK_BOUNDED')
        return s.st_mode,s.st_dev,s.st_ino,s.st_size,s.st_mtime_ns,s.st_nlink
    def once(prior=None):
        a=metadata(); before=identity(a)
        if prior is not None: need((before,a.st_ctime_ns)==prior,'FILE_CHANGED_DURING_READ')
        with path.open('rb') as f:
            h=os.fstat(f.fileno()); need(identity(h)==before,'OPEN_IDENTITY')
            raw=f.read(8*1024*1024+1); z=os.fstat(f.fileno())
            need(identity(z)==before and h.st_ctime_ns==z.st_ctime_ns,'HANDLE_CHANGED_DURING_READ')
        end=metadata(); need(identity(end)==before and end.st_ctime_ns==a.st_ctime_ns and len(raw)==a.st_size,'PATH_CHANGED_DURING_READ')
        need(sh(raw)==wanted,'FILE_BYTES'); return raw,(before,a.st_ctime_ns)
    a,identity_a=once(); b,identity_b=once(identity_a)
    need(a==b and identity_a==identity_b,'CLOSED_READ_BYTES_CHANGED'); return b

def _literal(source, name):
    """Parse an assignment's literal value; never compile or execute source."""
    try:
        tree=ast.parse(source.decode('utf8'))
        found=[n.value for n in tree.body if isinstance(n,ast.Assign) and any(isinstance(t,ast.Name) and t.id==name for t in n.targets)]
        need(len(found)==1,'SOURCE_LITERAL_REQUIRED'); value=ast.literal_eval(found[0]); typed(value); return value
    except EvidenceFailure: raise
    except Exception: raise EvidenceFailure('SOURCE_LITERAL_REJECTED') from None

def _frame(value):
    if type(value)is dict: return {'object':{k:_frame(v) for k,v in sorted(value.items())}}
    if type(value)is list: return {'array':[json.loads(x) for x in sorted({canonical(_frame(v)) for v in value})],'length':len(value)}
    return type(value).__name__

def _acl_rows(expected):
    rows=[]
    for entry in expected['nonowner_acl']:
        for column in (entry.get('columns') or [None]):
            rows.append({'kind':entry['kind'],'object':entry['object'],'column':column,'grantee':entry['grantee'],'grantor':entry['grantor'].replace('<original-bootstrap-grantor>','postgres'),'privilege':entry['privilege'],'grant_option':entry['grant_option']})
    # The schema owner has implicit ownership rights, not nonowner ACL entries.
    return [r for r in rows if not (r['kind']=='SCHEMA' and (r['object']=='public' or r['grantee']==expected['schemas'][r['object']]['owner']))]

def _has(expected, role, kind, name, privilege, column=None):
    rows=_acl_rows(expected)
    owner=expected['schemas'].get(name,{}).get('owner') if kind=='SCHEMA' else expected['tables'].get(name,{}).get('owner') if kind=='TABLE' else expected['functions'].get(name,{}).get('owner')
    if role==owner: return True
    return any(r['kind']==kind and r['object']==name and r['grantee'] in (role,'PUBLIC') and r['privilege']==privilege and (r['column']is None or column is not None and r['column']==column) for r in rows)

def _metadata(case, descriptor, expected):
    """Independently rebuild the finite source query and expected ACL projection."""
    q=lambda x:"'"+x.replace("'","''")+"'"
    role=case['role']; kind=case['diagnostic']['kind']; schema=case['diagnostic'].get('schema'); requirement=descriptor['requirement']; profile=descriptor['profile']; values={}; terms={}
    def add(k,v,text): values[k]=v; terms[k]=text
    def right(k,n,v,c=None): return _has(expected,role,k,n,v,c)
    def table(n,v): return 'has_table_privilege('+q(role)+','+q(n)+','+q(v)+')'
    def column(n,c,v): return 'has_column_privilege('+q(role)+','+q(n)+','+q(c)+','+q(v)+')'
    if kind=='IMMUTABLE_GRANT': return None,{}
    if kind=='SET_ROLE': add('role_assumption_allowed',False,'pg_has_role('+q(role)+','+q(case['diagnostic']['target_role'])+",'SET')")
    else:
        add('schema_usage',right('SCHEMA',schema,'USAGE'),'has_schema_privilege('+q(role)+','+q(schema)+",'USAGE')")
        if kind=='SCHEMA_CREATE': add('schema_create',right('SCHEMA',schema,'CREATE'),'has_schema_privilege('+q(role)+','+q(schema)+",'CREATE')")
        elif kind=='FUNCTION':
            matches=[n for n,v in expected['functions'].items() if v['name']==schema+'.'+case['diagnostic']['function']]; need(len(matches)==1,'SOURCE_FUNCTION_IDENTITY')
            add('function_execute',right('FUNCTION',matches[0],'EXECUTE'),'has_function_privilege('+q(role)+','+q(matches[0])+",'EXECUTE')")
        elif profile!='PROVIDER_SCHEMA_ONLY_FUNCTION_CALL':
            name=requirement['table']; verb=requirement.get('operation','SELECT')
            add('table_privilege',right('TABLE',name,verb),table(name,verb))
            for key,cols,priv in (('write_columns',requirement.get('write_columns',[]),verb),('read_columns',requirement.get('read_columns',[]),'SELECT')):
                if cols: add(key,{c:right('TABLE',name,priv,c) for c in cols},'jsonb_build_object('+','.join(q(c)+','+column(name,c,priv) for c in cols)+')')
            if requirement.get('read_columns'): add('read_table_privilege',right('TABLE',name,'SELECT'),table(name,'SELECT'))
            if requirement.get('SELECT_any_actual_column'):
                add('any_column_select',any(right('TABLE',name,'SELECT',c) for c in expected['tables'][name]['columns']),'has_any_column_privilege('+q(role)+','+q(name)+",'SELECT')")
    query='SELECT jsonb_build_object('+','.join(q(k)+','+v for k,v in terms.items())+');'
    result=copy.deepcopy(values); missing=[]
    write_missing=bool(requirement.get('write_columns')) and not(values['table_privilege'] or all(values['write_columns'].values()))
    read_missing=bool(requirement.get('read_columns')) and not(values['read_table_privilege'] or all(values['read_columns'].values()))
    if requirement.get('SELECT_any_actual_column'): read_missing=not values['any_column_select']
    mandatory=bool(requirement.get('mandatory_table_privileges')) and not values['table_privilege']
    for key,table_key in (('write_columns','table_privilege'),('read_columns','read_table_privilege')):
        if key in values and not values[table_key]: missing.extend(c for c,v in values[key].items() if v is False)
    if 'table_privilege'in values:
        result.update(missing_referenced_columns=list(dict.fromkeys(missing)),required_permission_missing=bool(write_missing or read_missing or mandatory),permission_components_missing={'write':bool(write_missing),'read':bool(read_missing),'mandatory_table':bool(mandatory)},column_privilege_projection_matches_source=True)
        if profile.startswith('PROVIDER_SCHEMA_ONLY_'): result.update(required_permission_missing=False,missing_referenced_columns=[])
    result.pop('write_columns',None); result.pop('read_columns',None)
    return sh(query.encode()),result

class QualificationBatch:
    """Private captured bytes, exact44 once, sticky refusal, final fresh519 read."""
    def __init__(self, root):
        path=Path(root); need(not path.is_symlink() and not getattr(path,'is_junction',lambda:False)(),'ROOT_LINK'); self.__root=path.resolve(strict=True)
        blobs=_capture(self.__root); self.__execution=validate_bundle(blobs); execution=doc(blobs[DERIVED])
        self.__rows=MappingProxyType({(c['case_id'],c['assertion_id']):(i,canonical(_expected_row(c,i))) for i,c in enumerate(execution['claims'])})
        need(len(self.__rows)==44,'EXACT44_BATCH_ROWS'); self.__bytes=MappingProxyType(blobs); self.__seen=set(); self.__failed=None; self.__finalized=False
    def _reject(self, code):
        if self.__failed is None: self.__failed=code
        raise EvidenceFailure(self.__failed) from None
    def validate_claim(self, case_id, assertion_id, status, execution, root=None, json_pointer=None, ledger_row=None):
        try:
            need(self.__failed is None and not self.__finalized,'BATCH_CLOSED'); need(all(type(x)is str for x in (case_id,assertion_id,status,json_pointer)),'CLAIM_TYPES')
            key=case_id,assertion_id; need(key in self.__rows and key not in self.__seen,'EXACT_UNSEEN44'); index,row=self.__rows[key]
            need(MAXIMA[key]==status and json_pointer=='/claims/'+str(index),'EXACT_STATUS_POINTER')
            if root is not None: need(Path(root).resolve(strict=True)==self.__root,'BATCH_ROOT')
            typed(execution); typed(ledger_row); need(type(execution)is dict and type(ledger_row)is dict and canonical(execution)==self.__execution and canonical(ledger_row)==row,'EXACT_EXECUTION_FULL_LEDGER_ROW')
            self.__seen.add(key); return True
        except EvidenceFailure as error: self._reject(str(error))
        except Exception: self._reject('PRIVILEGE485517B0_EVIDENCE_REJECTED')
    def verify_unchanged(self):
        try:
            need(self.__failed is None and not self.__finalized,'BATCH_CLOSED'); need(self.__seen==set(self.__rows),'COMPLETE44_BATCH_REQUIRED')
            fresh=_capture(self.__root); need(all(fresh[p]==self.__bytes[p] for p in PINS),'FINAL_ALL_INPUT_BYTES_DRIFT'); self.__bytes=None; self.__finalized=True; return True
        except EvidenceFailure as error: self._reject(str(error))
        except Exception: self._reject('PRIVILEGE485517B0_EVIDENCE_REJECTED')
    def require_finalized(self):
        if self.__failed is not None: raise EvidenceFailure(self.__failed)
        if not self.__finalized: self._reject('FINAL_FRESH_ALL_INPUT_BARRIER_REQUIRED')
        return True
