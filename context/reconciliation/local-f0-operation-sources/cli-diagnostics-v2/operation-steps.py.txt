"""Literal local migration experiment steps. Loaded only after snapshot hash check.

No statement bodies, credentials, raw handles or unrestricted process messages
enter retained evidence. One expected setup fault; every unexpected result stops.
"""
import runpy
import tarfile

def validate_archive(path,expected):
    seen=set()
    with tarfile.open(path,'r:gz') as archive:
        for member in archive:
            need(member.isfile() and not member.name.startswith('/') and all(p not in ('','.','..') for p in member.name.split('/')) and member.name not in seen,'UNSAFE_ARCHIVE_MEMBER')
            need(member.name in expected,'UNLISTED_ARCHIVE_MEMBER')
            need(hashlib.sha256(archive.extractfile(member).read()).hexdigest()==expected[member.name],'ARCHIVE_MEMBER_HASH_DRIFT')
            seen.add(member.name)
    need(seen==set(expected),'ARCHIVE_FILE_SET_MISMATCH')

def verify_packet():
    need(MANIFEST['review_status']=='APPROVED_GENUINE_MIGRATION_ATTEMPT','OPERATION_NOT_REVIEWED')
    need(MANIFEST['wrapper_sha256']==hashlib.sha256(Path(__file__).read_bytes()).hexdigest(),'WRAPPER_SOURCE_MISMATCH')
    need(command(['git','-c','core.longpaths=true','rev-parse','HEAD']).stdout.strip()==COMMIT and command(['git','-c','core.longpaths=true','rev-parse','HEAD^{tree}']).stdout.strip()==TREE,'SOURCE_CHECKPOINT_MISMATCH')
    hashes=MANIFEST['packet_files_sha256']
    required={'review-output/foundation-bootstrap.review.sql','review-output/foundation-bootstrap-finalize.review.sql','review-output/foundation-application.review.sql','scripts/sql-review/run-first-rehearsal.py','scripts/sql-review/rehearsal-diagnostics.py','scripts/prepare-local-f0-history.py','scripts/sql-review/run-local-f0-migrate.py','prisma.local-f0.config.mjs','scripts/sql-review/local-f0-migrate-guard.mjs','scripts/sql-review/local-f0-migrate-runner.mjs'}
    need(required<=set(hashes),'REQUIRED_PACKET_FILES_MISSING')
    for path,expected in hashes.items():
        need(not Path(path).is_absolute() and '..' not in Path(path).parts,'UNSAFE_PACKET_PATH')
        need(hashlib.sha256((ROOT/path).read_bytes()).hexdigest()==expected,'PACKET_FILE_DRIFT',{'path':path})
        blob=subprocess.run(['git','-c','core.longpaths=true','show',COMMIT+':'+path],cwd=ROOT,capture_output=True,timeout=30)
        need(blob.returncode==0 and hashlib.sha256(blob.stdout).hexdigest()==expected,'COMMITTED_PACKET_MISMATCH',{'path':path})
    aggregate=hashlib.sha256(''.join(p+'\0'+hashes[p]+'\n' for p in sorted(hashes)).encode()).hexdigest()
    need(aggregate==PACKET,'PACKET_AGGREGATE_MISMATCH')
    need(MANIFEST['image']==IMAGE and MANIFEST['database']==DB and MANIFEST['run_label']==LABEL,'TARGET_MANIFEST_MISMATCH')
    need(hashlib.sha256(Path(MANIFEST['tool_archive']).read_bytes()).hexdigest()==MANIFEST['tool_archive_sha256'],'TOOL_ARCHIVE_DRIFT')
    verified_tools=None
    for kind,plan in MANIFEST['migration_targets'].items():
        raw=Path(plan['manifest_path']).read_bytes();need(hashlib.sha256(raw).hexdigest()==plan['manifest_sha256'],'TARGET_PLAN_DRIFT')
        migration=json.loads(raw);target=next(t for t in TARGETS if t['kind']==kind)
        if verified_tools is None:
            validate_archive(MANIFEST['tool_archive'],migration['tools']['files_sha256']);verified_tools=migration['tools']['files_sha256']
        else:need(verified_tools==migration['tools']['files_sha256'],'TARGET_TOOL_CLOSURES_DIFFER')
        need(migration['run_id']==RUN_ID and migration['target']['container_name']==target['name'] and migration['target']['volume_name']==target['volume'],'TARGET_PLAN_IDENTITY')
        need(migration['target']['bootstrap_secret_host_path']==str(PRIVATE/(kind+'-bootstrap-password')),'SECRET_MOUNT_PLAN_MISMATCH')
        snapshot=Path(plan['source_snapshot'])
        current={p.relative_to(snapshot).as_posix():hashlib.sha256(p.read_bytes()).hexdigest() for p in snapshot.rglob('*') if p.is_file()}
        need(current==plan['source_files_sha256'],'SOURCE_SNAPSHOT_DRIFT')
        need(hashlib.sha256(Path(plan['source_archive']).read_bytes()).hexdigest()==plan['source_archive_sha256'],'SOURCE_ARCHIVE_DRIFT')
        validate_archive(plan['source_archive'],current)
        for path in ('prisma.local-f0.config.mjs','scripts/sql-review/local-f0-migrate-guard.mjs','scripts/sql-review/local-f0-migrate-runner.mjs','prisma/review/foundation-candidate.prisma','prisma/foundation/migrations/migration_lock.toml','prisma/foundation/migrations/20261117000100_f0_local_initial/migration.sql'):
            need(current[path]==hashes[path],'SOURCE_SNAPSHOT_NOT_COMMITTED_PACKET',{'path':path})
    report['source_verification']={'commit':COMMIT,'tree':TREE,'packet_sha256':aggregate,'files_sha256':hashes,'file_count':len(hashes)}
    report['run_manifest']=MANIFEST;report['run_manifest_sha256']=hashlib.sha256(MANIFEST_PATH.read_bytes()).hexdigest()
    event('SOURCE_VERIFICATION','PASS',files=len(hashes),packet_sha256=aggregate)

def migrate(target,operation,label,proof=None):
    name=target['name'];plan=MANIFEST['migration_targets'][target['kind']]
    module=runpy.run_path(str(ROOT/'scripts/sql-review/run-local-f0-migrate.py'))
    report['stage']=label;save()
    def created_identity_process(arguments,**options):
        result=subprocess.run(arguments,**options)
        if arguments==['docker','--context','desktop-linux','inspect',name] and result.returncode==0:
            objects=json.loads(result.stdout)
            need(len(objects)==1 and objects[0]['Id']==created_containers[name],'CREATED_CONTAINER_IDENTITY_CHANGED')
        return result
    try:value=module['coordinate'](Path(plan['manifest_path']).read_bytes(),operation,None if proof is None else proof.read_bytes(),process=created_identity_process)
    except module['HeldFailure'] as error:
        comparison=module['mount_failure_comparison'](error)
        if comparison is not None:report['resource_guard_comparison']=comparison;save()
        raise Stop('MIGRATION_GUARD_REFUSED',{'code':error.code,**({'mount_comparison':comparison} if comparison is not None else {})}) from None
    except subprocess.TimeoutExpired:raise Stop('MIGRATION_PROCESS_TIMEOUT_CLOSURE_UNCONFIRMED') from None
    except Exception:raise Stop('MIGRATION_RESPONSE_OR_PROCESS_UNCONFIRMED') from None
    report.setdefault('migration_operations',[]).append({'target':name,'stage':label,**value});save()
    return value

def ledger(name,stage):
    # Logs never leave PostgreSQL: only allowlisted exact-header comparisons.
    presence=observe(name,"""SELECT jsonb_build_object(
 'ledger_present',EXISTS(SELECT 1 FROM pg_class c JOIN pg_namespace n ON n.oid=c.relnamespace WHERE n.nspname='public' AND c.relname='_prisma_migrations'),
 'ledger_regular',EXISTS(SELECT 1 FROM pg_class c JOIN pg_namespace n ON n.oid=c.relnamespace WHERE n.nspname='public' AND c.relname='_prisma_migrations' AND c.relkind='r'),
 'ledger_owner_expected',EXISTS(SELECT 1 FROM pg_class c JOIN pg_namespace n ON n.oid=c.relnamespace WHERE n.nspname='public' AND c.relname='_prisma_migrations' AND c.relowner='wa_migrator'::regrole),
 'public_relation_count',(SELECT count(*) FROM pg_class c JOIN pg_namespace n ON n.oid=c.relnamespace WHERE n.nspname='public' AND c.relkind='r'),
 'application_relations',(SELECT count(*) FROM pg_class c JOIN pg_namespace n ON n.oid=c.relnamespace WHERE n.nspname IN ('app','platform','platform_private')),
 'application_tables',(SELECT count(*) FROM pg_class c JOIN pg_namespace n ON n.oid=c.relnamespace WHERE n.nspname IN ('app','platform','platform_private') AND c.relkind='r'),
 'application_functions',(SELECT count(*) FROM pg_proc p JOIN pg_namespace n ON n.oid=p.pronamespace WHERE n.nspname IN ('app','platform','platform_private')),
 'application_types',(SELECT count(*) FROM pg_type t JOIN pg_namespace n ON n.oid=t.typnamespace WHERE n.nspname IN ('app','platform','platform_private')),
 'application_enums',(SELECT count(*) FROM pg_type t JOIN pg_namespace n ON n.oid=t.typnamespace WHERE n.nspname IN ('app','platform','platform_private') AND t.typtype='e'),
 'active_migrators',(SELECT count(*) FROM pg_stat_activity WHERE datname=current_database() AND usename='wa_migrator'));""",stage+'_CATALOG_PRESENCE')
    retained={'stage':stage,**presence,'rows':None,'ledger_rows_observed':False}
    report['targets'][name].setdefault('ledger_observations',[]).append(retained);save()
    need(presence['ledger_present'] and presence['ledger_regular'] and presence['ledger_owner_expected'] and presence['active_migrators']==0,'PRIVATE_LEDGER_PRESENCE_OR_BACKEND_BOUNDARY_FAILED')
    value=observe(name,"""SELECT jsonb_build_object(
 'owner',(SELECT pg_get_userbyid(relowner) FROM pg_class c JOIN pg_namespace n ON n.oid=c.relnamespace WHERE n.nspname='public' AND c.relname='_prisma_migrations'),
 'public_relation_count',(SELECT count(*) FROM pg_class c JOIN pg_namespace n ON n.oid=c.relnamespace WHERE n.nspname='public' AND c.relkind='r'),
 'runtime_ledger_access',has_table_privilege('wa_app_runtime','public._prisma_migrations','SELECT,INSERT,UPDATE,DELETE') OR has_table_privilege('wa_worker','public._prisma_migrations','SELECT,INSERT,UPDATE,DELETE') OR has_table_privilege('wa_context_issuer','public._prisma_migrations','SELECT,INSERT,UPDATE,DELETE'),
 'application_relations',(SELECT count(*) FROM pg_class c JOIN pg_namespace n ON n.oid=c.relnamespace WHERE n.nspname IN ('app','platform','platform_private')),
 'application_tables',(SELECT count(*) FROM pg_class c JOIN pg_namespace n ON n.oid=c.relnamespace WHERE n.nspname IN ('app','platform','platform_private') AND c.relkind='r'),
 'application_functions',(SELECT count(*) FROM pg_proc p JOIN pg_namespace n ON n.oid=p.pronamespace WHERE n.nspname IN ('app','platform','platform_private')),
 'application_types',(SELECT count(*) FROM pg_type t JOIN pg_namespace n ON n.oid=t.typnamespace WHERE n.nspname IN ('app','platform','platform_private')),
 'application_enums',(SELECT count(*) FROM pg_type t JOIN pg_namespace n ON n.oid=t.typnamespace WHERE n.nspname IN ('app','platform','platform_private') AND t.typtype='e'),
 'active_migrators',(SELECT count(*) FROM pg_stat_activity WHERE datname=current_database() AND usename='wa_migrator'),
 'rows',(SELECT coalesce(jsonb_agg(jsonb_build_object('name',migration_name,'checksum',checksum,'finished',finished_at IS NOT NULL,'rolled_back',rolled_back_at IS NOT NULL,'steps',applied_steps_count,
 'expected_error_code',coalesce(logs,'') ~ E'(^|\\n)Database error code: 42501(\\r?\\n|$)',
 'expected_set_role_error',coalesce(logs,'') ~ E'(^|\\n)ERROR: must be able to SET ROLE "wa_customer_verifier"(\\r?\\n|$)') ORDER BY started_at),'[]'::jsonb) FROM public._prisma_migrations));""",stage)
    retained.update(value,ledger_rows_observed=True);save()
    need(value['owner']=='wa_migrator' and value['public_relation_count']==1 and not value['runtime_ledger_access'] and value['active_migrators']==0,'PRIVATE_LEDGER_OR_BACKEND_BOUNDARY_FAILED')

    need(all(row['name']==intended['migration']['id'] and row['checksum']==intended['source']['integrated_sql_sha256'] for row in value['rows']),'LEDGER_IDENTITY_OR_CHECKSUM_MISMATCH')

def attempt_expected_failure(target):
    name=target['name']
    report['steps']['MIGRATE-FAILURE']='ATTEMPTED_UNQUALIFIED';save()
    try:
        result=migrate(target,'deploy','EXPECTED_FAILED_PRISMA_DEPLOY')
    finally:finalize(name)
    # A returned strict host result proves the process exited. Observe its
    # database state before testing codes; an unknown failure must keep clues.
    need(result['status'] in ('CLI_FAILED_STOP_AND_INSPECT','CLI_OUTPUT_REJECTED_STOP_AND_INSPECT') and type(result['exit_code']) is int and result['exit_code']!=0,'EXPECTED_CLOSED_CLI_FAILURE_NOT_PROVEN')
    failed=ledger(name,'FAILED_DEPLOY_LEDGER_AND_ROLLBACK')
    need(result['status']=='CLI_FAILED_STOP_AND_INSPECT' and result['prisma_codes']==['P3018'] and result['sqlstates']==['42501'],'EXPECTED_MIGRATION_FAILURE_NOT_PROVEN')
    return failed

def migrate_success(target,operation,label,proof=None):
    value=migrate(target,operation,label,proof)
    need(value['status']=='CLI_COMPLETED_REQUIRES_INDEPENDENT_LEDGER_CATALOG_CHECKS' and type(value['exit_code']) is int and value['exit_code']==0,'PRISMA_COMPLETED_STATUS_REQUIRED')
    return value

def schema_fingerprint(name):
    args=['exec','-e','PGPASSFILE=/run/wa-secrets/pgpass',created_containers[name],'pg_dump','--schema-only','--schema=app','--schema=platform','--schema=platform_private','--schema=public','-h','127.0.0.1','-U','postgres','-d',DB]
    result=docker(args,timeout=120);need(result.returncode==0,'NATIVE_SCHEMA_FINGERPRINT_FAILED')
    lines=result.stdout.splitlines();meta=[line for line in lines if line.startswith(('\\restrict ','\\unrestrict '))]
    need(len(meta) in (0,2),'UNEXPECTED_DUMP_META_SHAPE')
    normalized='\n'.join(line for line in lines if not line.startswith(('\\restrict ','\\unrestrict ')))+'\n'
    return {'sha256':hashlib.sha256(normalized.encode()).hexdigest(),'random_dump_meta_lines_removed':len(meta),'raw_schema_dump_retained':False}
