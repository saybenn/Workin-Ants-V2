class Refused(Exception): pass
def need(value,code):
    if not value:raise Refused(code)
def sha(raw):return hashlib.sha256(raw).hexdigest()
def data(value):return (json.dumps(value,indent=2,allow_nan=False)+'\n').encode()


def prepare(baseline_path,baseline_sha,run_id=None):
    baseline_path=Path(baseline_path).resolve()
    need(baseline_path.is_relative_to(TASK) and not baseline_path.is_relative_to(ROOT),'EXTERNAL_BASELINE_PATH')
    raw=baseline_path.read_bytes();need(sha(raw)==baseline_sha,'REVIEWED_BASELINE_SHA')
    baseline=json.loads(raw)
    need(baseline['purpose']=='LOCAL_F0_NATIVE_CATALOG_REHEARSAL' and baseline['image']==IMAGE and baseline['database']=='wa_p0c_rehearsal','BASELINE_SCOPE')
    proposal_raw=(ROOT/PROPOSAL).read_bytes();need(sha(proposal_raw)==PROPOSAL_SHA,'APPROVED_RESTORE_PROPOSAL')
    proposal=json.loads(proposal_raw);need(proposal['source_only_packet_sha256']==CONTENT_SHA and proposal['status']=='NOT_RUN','RESTORE_PROPOSAL_SCOPE')
    for name,expected in proposal['source_files_sha256'].items():need(sha((ROOT/name).read_bytes())==expected,'RESTORE_PROPOSAL_INPUT_DRIFT')
    for name,expected in proposal['prepared_files_sha256'].items():need(sha((ROOT/'review-output/restore-control-rehearsal'/name).read_bytes())==expected,'RESTORE_ARTIFACT_DRIFT')
    for artifact in baseline['catalog']['assets'].values():need(sha((ROOT/artifact['path']).read_bytes())==artifact['sha256'],'BASELINE_CATALOG_ASSET_DRIFT')
    need(sha((ROOT/SQL).read_bytes())==sha((ROOT/MIGRATION).read_bytes())==SQL_SHA,'CANONICAL_SQL')
    identity=str(uuid.uuid4()) if run_id is None else run_id
    need(type(identity) is str and str(uuid.UUID(identity))==identity and uuid.UUID(identity).version==4,'FRESH_RUN_UUID4')
    stem=identity[:8];name='wa-p0c-'+stem+'-restore-source';target='wa-p0c-'+stem+'-restore-target';stage='wa-p0c-'+stem+'-restore-staging'
    directory=TASK/('local-f0-restore-plan-'+stem);out=TASK/('local-f0-restore-'+stem)
    need(not directory.exists() and not out.exists(),'FRESH_PLAN_AND_EVIDENCE_REQUIRED')
    old=baseline['migration_targets']['ok'];old_raw=Path(old['manifest_path']).read_bytes();need(sha(old_raw)==old['manifest_sha256'],'BASELINE_NESTED_PLAN')
    nested=json.loads(old_raw)
    nested['run_id']=identity;nested['target'].update(container_name=name,volume_name=name+'-data',bootstrap_secret_host_path=str(out/'private/ok-bootstrap-password'))
    nested['target']['labels']={'workinants.rehearsal':'p0c-local-f0','workinants.rehearsal.run':identity}
    nested['source']['runtime_files_sha256']={path:sha((ROOT/path).read_bytes()) for path in RUNTIME_FILES}
    need(nested['source']['integrated_sql_sha256']==SQL_SHA and nested['source']['foundation_schema_sha256']==sha((ROOT/SCHEMA).read_bytes()),'BASELINE_SOURCE_BINDING')
    need(sha(Path(baseline['tool_archive']).read_bytes())==baseline['tool_archive_sha256'],'BASELINE_TOOL_ARCHIVE')
    closure=json.loads((ROOT/'context/reconciliation/local-f0-tooling/prisma-tools-closure.json').read_bytes())
    need(nested['tools']['files_sha256']==closure['tools_files_sha256'],'TOOL_CLOSURE_BINDING')
    snapshot={path:(ROOT/path).read_bytes() for path in (MIGRATION,LOCK,SCHEMA,*RUNTIME_FILES)}
    history_raw=(Path(old['source_snapshot'])/'prisma/foundation/history-manifest.json').read_bytes()
    need(sha(history_raw)==old['source_files_sha256']['prisma/foundation/history-manifest.json'],'BASELINE_HISTORY_SNAPSHOT')
    history=json.loads(history_raw)
    need(history['source_sql_sha256']==SQL_SHA and history['files_sha256']=={path:sha(value) for path,value in snapshot.items()},'EXISTING_HISTORY_SOURCE_ONLY')
    nested_raw=data(nested);history['run_id']=identity;history['run_manifest_sha256']=sha(nested_raw)
    snapshot['prisma/foundation/history-manifest.json']=data(history)
    master=copy.deepcopy(baseline)
    master.update(purpose='LOCAL_F0_RESTORE_REHEARSAL',run_id=identity,source_commit='TO_BE_PINNED',source_tree='TO_BE_PINNED',packet_files_sha256={},packet_sha256='TO_BE_PINNED',
        evidence_directory=str(out),targets=[{'name':name,'volume':name+'-data','kind':'ok'}],review_status='PENDING_OPERATIONAL_REVIEW',status='SOURCE_PREPARED_NOT_EXECUTED')
    master['migration_targets']={'ok':{'manifest_path':str(directory/'ok-migration-manifest.json'),'manifest_sha256':sha(nested_raw),'source_snapshot':str(directory/'source-ok'),
        'source_files_sha256':{path:sha(value) for path,value in snapshot.items()},'source_archive':str(directory/'source-ok.tar.gz')}}
    external={path:sha((TASK/path).read_bytes()) for path in EXTERNAL_FILES}
    master['wrapper_sha256']=external[EXTERNAL_FILES[0]];master['steps_sha256']=external[EXTERNAL_FILES[1]]

        'archive_path':str(out/'private/restore01-base.tar'),'external_files_sha256':external}

    master['storage']=copy.deepcopy(proposal['required_literal_resource_manifest'])

            for path,value in sorted(snapshot.items()):

    source_archive=archive.getvalue();(directory/'source-ok.tar.gz').write_bytes(source_archive)

    return {'status':'PREPARED_SOURCE_ONLY_HELD','run_id':identity,'operation_manifest':str(directory/'operation-manifest.json'),'operation_manifest_sha256':sha(data(master)),

    args=parser.parse_args(argv)
    try:
        need(args.prepare_plan and args.baseline_manifest is not None and args.baseline_sha256 is not None,'EXPLICIT_REVIEWED_BASELINE_REQUIRED')
        value=prepare(args.baseline_manifest,args.baseline_sha256,args.run_id);print(json.dumps(value));return 0
    except BaseException as error:
        print(json.dumps({'status':'HELD','code':str(error) if isinstance(error,Refused) else 'SOURCE_PREPARATION_REFUSED','raw_output_discarded':True}));return 1
