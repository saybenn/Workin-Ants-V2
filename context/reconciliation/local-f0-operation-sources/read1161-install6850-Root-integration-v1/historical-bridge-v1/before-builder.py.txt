"""Merge exact selected assertions with separately retained execution evidence.

The inventory remains a reproducible plan. This derived board never substitutes
dependency denials for missing positives or silently promotes compound cases.
"""
import hashlib,json
from pathlib import Path
from local_f0_migration_evidence import validate_claim, EvidenceFailure
from local_f0_catalog_evidence import validate_claim as validate_catalog_claim, EvidenceFailure as CatalogEvidenceFailure
from local_f0_restricted_evidence import validate_claim as validate_restricted_claim, EvidenceFailure as RestrictedEvidenceFailure
from local_f0_native_prefix_evidence import validate_claim as validate_native_prefix_claim, EvidenceFailure as NativePrefixEvidenceFailure
from local_f0_owner_lease_evidence import validate_claim as validate_owner_lease_claim, EvidenceFailure as OwnerLeaseEvidenceFailure
from local_f0_native_complete_evidence import prepare_batch as prepare_native_complete_batch, EvidenceFailure as NativeCompleteEvidenceFailure
from local_f0_scoped_read_evidence import prepare_batch as prepare_scoped_read_batch, EvidenceFailure as ScopedReadEvidenceFailure
from local_f0_install_evidence import prepare_batch as prepare_install_batch, EvidenceFailure as InstallEvidenceFailure
from local_f0_worker_recovery_evidence_v2 import prepare_batch as prepare_worker_recovery_batch, EvidenceFailure as WorkerRecoveryEvidenceFailure
from local_f0_read1161_evidence import prepare_batch as prepare_read1161_batch, EvidenceFailure as Read1161EvidenceFailure
from local_f0_install_precleanup_evidence import prepare_batch as prepare_install_precleanup_batch, EvidenceFailure as InstallPrecleanupEvidenceFailure
ROOT=Path(__file__).resolve().parents[1]
REC=ROOT/'context/reconciliation'
SMALL_PASSES={(f'P0C-TABLE-{model}-ALLOW','positive.wa_app_runtime.insert') for model in ('CustomerProfile','CustomerProfileLifecycleEvent','RuntimeCommandReceipt','OwnerMutationEvidence','RuntimeOutboxMessage')}|{
    ('P0C-TABLE-CustomerProfile-DENY','denial.context.foreign_runtime_context'),
    ('P0C-BOOT-01','control.nonrecursive_create'),
    ('P0C-BOOT-01','vector.FIRST-SMALL-REHEARSAL.positive'),
    ('P0C-BOOT-01','vector.FIRST-SMALL-REHEARSAL.adversarial'),
    ('P0C-ATOMIC-01','control.complete_group'),
    ('P0C-ATOMIC-01','vector.MESSAGE-REVERSE.positive'),
    ('P0C-ATOMIC-01','vector.RESULT-VERSION.positive'),
    ('P0C-INSTALL-01','control.success_cleanup'),
    ('P0C-INSTALL-02','control.failed_cleanup'),
    ('P0C-INSTALL-02','vector.FUNCTION-ACL-FAILURE-CLEANUP.positive'),
}
EVIDENCE_RECORDS={
    'context/reconciliation/local-f0-read13-0ff2a91-116153b2-qualified-evidence.json':('c1a93ad627e2276b6eceaf0f863efff24890a5c581fbb19a7c4a5c158ede2451','DERIVED_REVIEWED_REMAINING13_SQL_SELECT_ASSERTIONS_ONLY'),
    'context/reconciliation/local-f0-install-0ff2a91-6850ee4c-precleanup-qualified-evidence.json':('912b8ed02988b58e5df0268b5c0038b987f265e658c730d6895ca8d947d45f76','DERIVED_REVIEWED_INSTALL6850_PRECLEANUP_ONE_ASSERTION_ONLY'),
    'context/reconciliation/local-f0-worker-5f189aa-b42c35d5-qualified-evidence.json':('45489b9495ca594034ee9a4d8651813826da464cb12b83b0f82641efb21c897a','DERIVED_REVIEWED_WORKER17_FINITE_SQL_ASSERTIONS_ONLY'),
    'context/reconciliation/local-f0-install-5f189aa-4b35e08b-qualified-evidence.json':('08f81059dfc828007cbf69ae08b224a6cbd82196800e30650058d128e83d8e39','DERIVED_REVIEWED_INSTALL17_FINITE_ASSERTIONS_ONLY'),
    'context/reconciliation/local-f0-six-read-5f189aa-c662f807-qualified-evidence.json':('07b49506e56918515816949e628569f89ba12066d2b7ecc2ecf60977bf92fe7f','DERIVED_REVIEWED_SCOPED6_SQL_READ_ASSERTIONS_ONLY'),
    'context/reconciliation/local-f0-native-1110fa4-4c894dba-qualified-evidence.json':('1586330d18837b060f8968f85747151c4239271ef1c3cb3caad276773ddb1538','DERIVED_REVIEWED_NATIVE520_DENIAL_ASSERTIONS_ONLY'),
    'context/reconciliation/local-f0-lease-da5ad66-a560e05b-qualified-evidence.json':('33a356710320b9ec5381085354b3ed0199e146685d12364c618e27cd7e9772bc','DERIVED_REVIEWED_LEASE8_FINITE_ASSERTIONS_ONLY'),
    'context/reconciliation/local-f0-owner-da5ad66-72934689-qualified-evidence.json':('0effdedc5c26449d81eb49fa3581f54d57c13eb6e7ac889ce355fc3518f63c24','DERIVED_REVIEWED_OWNER20_FINITE_ASSERTIONS_ONLY'),
    'context/reconciliation/local-f0-small-rehearsal-0623028-execution.json':('269812ab154775e0cea58c3305ffbef141c2c0978590f8000c18c79cb2f0d28b','PASS_FIRST01_THROUGH_FIRST10_SUBSET_ONLY'),
    'context/reconciliation/local-f0-migration-6b8910d-execution.json':('e657bd4de3c288acbd86faa7ae9514f59e76566e1bba170a6ef19f7770c91158','PASS_GENUINE_FIRST_NOOP_FAILURE_REVIEWED_RESOLVE_SAME_CHECKSUM_RETRY'),
    'context/reconciliation/local-f0-catalog-b9703d7-23bd8979-execution.json':('bbf604e3eb2c7a066f2287a2f2f2ab825582d366dc5330394170b33a0fd22de8','PASS_LOCAL_NATIVE_CATALOG_SUBSET_ONLY'),
    'context/reconciliation/local-f0-restricted-b8f6b10-f161d454-execution.json':('9a090022c11f150cb133617af11cb9c6afafba1624313673ee3a5237ae637232','PASS_LOCAL_RESTRICTED_CONTROL_PACKETS_ONLY'),
    'context/reconciliation/local-f0-native-access-0a29cf6-495976eb-execution.json':('e47ecfcc00064e22f2731c6a1e24ddc64df672690c2eabbb821d45c3aa5a6bdf','STOPPED_ATTEMPT_PRESERVED'),
}

def build():
    inventory=json.loads((REC/'local-f0-assertions.json').read_text(encoding='utf8'))
    ledger=json.loads((REC/'local-f0-assertion-evidence.json').read_text(encoding='utf8'))
    read1161_batch=None
    if any(row.get('evidence_path')=='context/reconciliation/local-f0-read13-0ff2a91-116153b2-qualified-evidence.json' for row in ledger['assertions']):
        try:read1161_batch=prepare_read1161_batch(ROOT)
        except Read1161EvidenceFailure as error:raise AssertionError(str(error)) from None
    install_precleanup_batch=None
    if any(row.get('evidence_path')=='context/reconciliation/local-f0-install-0ff2a91-6850ee4c-precleanup-qualified-evidence.json' for row in ledger['assertions']):
        try:install_precleanup_batch=prepare_install_precleanup_batch(ROOT)
        except InstallPrecleanupEvidenceFailure as error:raise AssertionError(str(error)) from None
    worker_recovery_batch=None
    if any(row.get('evidence_path')=='context/reconciliation/local-f0-worker-5f189aa-b42c35d5-qualified-evidence.json' for row in ledger['assertions']):
        try:worker_recovery_batch=prepare_worker_recovery_batch(ROOT)
        except WorkerRecoveryEvidenceFailure as error:raise AssertionError(str(error)) from None
    install_batch=None
    if any(row.get('evidence_path')=='context/reconciliation/local-f0-install-5f189aa-4b35e08b-qualified-evidence.json' for row in ledger['assertions']):
        try:install_batch=prepare_install_batch(ROOT)
        except InstallEvidenceFailure as error:raise AssertionError(str(error)) from None
    scoped_read_batch=None
    if any(row.get('evidence_path')=='context/reconciliation/local-f0-six-read-5f189aa-c662f807-qualified-evidence.json' for row in ledger['assertions']):
        try:scoped_read_batch=prepare_scoped_read_batch(ROOT)
        except ScopedReadEvidenceFailure as error:raise AssertionError(str(error)) from None
    native_complete_batch=None
    if any(row.get('evidence_path')=='context/reconciliation/local-f0-native-1110fa4-4c894dba-qualified-evidence.json' for row in ledger['assertions']):
        try:native_complete_batch=prepare_native_complete_batch(ROOT)
        except NativeCompleteEvidenceFailure as error:raise AssertionError(str(error)) from None
    for source,expected in inventory['source_provenance']['files_sha256'].items():
        assert hashlib.sha256((ROOT/source).read_bytes()).hexdigest()==expected,'Inventory source changed; regenerate and review evidence applicability'
    evidence={(row['case_id'],row['assertion_id']):row for row in ledger['assertions']}
    assert len(evidence)==len(ledger['assertions']),'Duplicate assertion evidence'
    identities={(case['id'],assertion['id']) for case in inventory['cases'] for assertion in case['assertions']}
    assert set(evidence)<=identities,'Unselected/unknown assertion evidence'
    cases=[];totals={s:0 for s in ('PASS','PARTIAL','FAILED','DEPENDENCY_BLOCKED','NOT_RUN','UNRESOLVED_SOURCE')}
    for case in inventory['cases']:
        assertions=[]
        for item in case['assertions']:
            row=evidence.get((case['id'],item['id']))
            status='DEPENDENCY_BLOCKED' if item['availability']=='DEPENDENCY_BLOCKED' else 'UNRESOLVED_SOURCE' if item['availability'] in ('SPECIFICATION_CONFLICT','SOURCE_PATH_REVIEW_REQUIRED') else 'NOT_RUN'
            if row:
                assert row['status'] in ('PASS','PARTIAL','FAILED')
                assert row['expected_assertion_sha256']==hashlib.sha256(item['expected_assertion'].encode()).hexdigest(),'Assertion requirement changed; review evidence mapping'
                assert row['evidence_path'] in EVIDENCE_RECORDS,'Unsupported execution evidence path requires reviewed parser'
                path=ROOT/row['evidence_path']
                expected_sha,expected_kind=EVIDENCE_RECORDS[row['evidence_path']]
                raw=path.read_bytes()
                assert hashlib.sha256(raw).hexdigest()==row['evidence_sha256']==expected_sha,'Execution evidence changed from independently reviewed record'
                execution=json.loads(raw)
                assert execution.get('actual_result')==expected_kind,'Execution evidence kind changed'
                for identity in ('run_id','source_commit','source_tree','packet_sha256'):
                    assert row[identity]==execution[identity],'Execution identity mismatch'
                pointer=row['json_pointer']
                assert type(pointer) is str and pointer.startswith('/'),'Evidence pointer invalid'
                value=execution
                for part in pointer[1:].split('/'):
                    part=part.replace('~1','/').replace('~0','~')
                    if isinstance(value,list):
                        assert part.isdigit() and str(int(part))==part and int(part)<len(value),'Evidence array pointer invalid'
                        value=value[int(part)]
                    else:
                        assert isinstance(value,dict) and part in value,'Evidence pointer missing'
                        value=value[part]
                assert type(value) is dict,'Evidence pointer must resolve to structured observation'
                if execution.get('actual_result')=='PASS_FIRST01_THROUGH_FIRST10_SUBSET_ONLY':
                    assert row['status']=='PASS' and (case['id'],item['id']) in SMALL_PASSES,'Unsupported small-rehearsal assertion promotion'
                elif execution.get('actual_result')=='PASS_LOCAL_NATIVE_CATALOG_SUBSET_ONLY':
                    expected_pointer='/sanitized_report/ledger_comparison' if (case['id'],item['id'])==('P0C-INSTALL-01','control.private_ledger') else '/sanitized_report/catalog_comparison'
                    assert pointer==expected_pointer,'Catalog evidence pointer does not identify the reviewed proof'
                    try:validate_catalog_claim(case['id'],item['id'],row['status'],execution,ROOT)
                    except CatalogEvidenceFailure as error:raise AssertionError(str(error)) from None
                elif execution.get('actual_result')=='PASS_LOCAL_RESTRICTED_CONTROL_PACKETS_ONLY':
                    try:validate_restricted_claim(case['id'],item['id'],row['status'],execution,ROOT,json_pointer=pointer)
                    except RestrictedEvidenceFailure as error:raise AssertionError(str(error)) from None
                elif row['evidence_path']=='context/reconciliation/local-f0-read13-0ff2a91-116153b2-qualified-evidence.json':
                    try:read1161_batch.validate_claim(case['id'],item['id'],row['status'],execution,ROOT,json_pointer=pointer,ledger_row=row)
                    except Read1161EvidenceFailure as error:raise AssertionError(str(error)) from None
                elif row['evidence_path']=='context/reconciliation/local-f0-worker-5f189aa-b42c35d5-qualified-evidence.json':
                    try:worker_recovery_batch.validate_claim(case['id'],item['id'],row['status'],execution,ROOT,json_pointer=pointer,ledger_row=row)
                    except WorkerRecoveryEvidenceFailure as error:raise AssertionError(str(error)) from None
                elif row['evidence_path']=='context/reconciliation/local-f0-install-0ff2a91-6850ee4c-precleanup-qualified-evidence.json':
                    try:install_precleanup_batch.validate_claim(case['id'],item['id'],row['status'],execution,ROOT,json_pointer=pointer)
                    except InstallPrecleanupEvidenceFailure as error:raise AssertionError(str(error)) from None
                elif row['evidence_path']=='context/reconciliation/local-f0-install-5f189aa-4b35e08b-qualified-evidence.json':
                    try:install_batch.validate_claim(case['id'],item['id'],row['status'],execution,ROOT,json_pointer=pointer)
                    except InstallEvidenceFailure as error:raise AssertionError(str(error)) from None
                elif row['evidence_path']=='context/reconciliation/local-f0-six-read-5f189aa-c662f807-qualified-evidence.json':
                    try:scoped_read_batch.validate_claim(case['id'],item['id'],row['status'],execution,ROOT,json_pointer=pointer)
                    except ScopedReadEvidenceFailure as error:raise AssertionError(str(error)) from None
                elif row['evidence_path']=='context/reconciliation/local-f0-native-1110fa4-4c894dba-qualified-evidence.json':
                    try:native_complete_batch.validate_claim(case['id'],item['id'],row['status'],execution,ROOT,json_pointer=pointer)
                    except NativeCompleteEvidenceFailure as error:raise AssertionError(str(error)) from None
                elif row['evidence_path'] in (
                    'context/reconciliation/local-f0-owner-da5ad66-72934689-qualified-evidence.json',
                    'context/reconciliation/local-f0-lease-da5ad66-a560e05b-qualified-evidence.json',
                ):
                    try:validate_owner_lease_claim(case['id'],item['id'],row['status'],execution,ROOT,json_pointer=pointer)
                    except OwnerLeaseEvidenceFailure as error:raise AssertionError(str(error)) from None
                elif row['evidence_path']=='context/reconciliation/local-f0-native-access-0a29cf6-495976eb-execution.json':
                    # A stopped run can qualify only its independently audited
                    # completed facets, with every original boundary intact.
                    try:validate_native_prefix_claim(case['id'],item['id'],row['status'],execution,ROOT,json_pointer=pointer)
                    except NativePrefixEvidenceFailure as error:raise AssertionError(str(error)) from None
                else:
                    try:validate_claim(case['id'],item['id'],row['status'],execution,ROOT)
                    except EvidenceFailure as error:raise AssertionError(str(error)) from None
                assert row['application_sql_sha256']==inventory['source_provenance']['files_sha256']['review-output/foundation-application.review.sql'],'SQL changed; retain old evidence but review qualification before reuse'
                assert not (status in ('DEPENDENCY_BLOCKED','UNRESOLVED_SOURCE') and row['status']=='PASS'),'Unavailable positive cannot be passed by denial'
                status=row['status']
            totals[status]+=1
            assertions.append({'id':item['id'],'mandatory':item['mandatory_for_case'],'status':status,'dependency_ids':item['dependency_ids'],**({'evidence':row} if row else {})})
        statuses=[a['status'] for a in assertions if a['mandatory']]
        state='PASS' if statuses and all(s=='PASS' for s in statuses) else 'FAILED' if 'FAILED' in statuses else 'PARTIAL' if any(s in ('PASS','PARTIAL') for s in statuses) else 'DEPENDENCY_BLOCKED' if all(s=='DEPENDENCY_BLOCKED' for s in statuses) else 'NOT_RUN'
        assert state!='PASS' or all(a['status']=='PASS' for a in assertions if a['mandatory'])
        cases.append({'id':case['id'],'status':state,'assertions':assertions})
    assert len(cases)==94 and sum(totals.values())==inventory['summary']['mandatory_assertions']
    if native_complete_batch is not None:
        try:native_complete_batch.verify_unchanged();native_complete_batch.require_finalized()
        except NativeCompleteEvidenceFailure as error:raise AssertionError(str(error)) from None
    if scoped_read_batch is not None:
        try:scoped_read_batch.verify_unchanged();scoped_read_batch.require_finalized()
        except ScopedReadEvidenceFailure as error:raise AssertionError(str(error)) from None
    if install_batch is not None:
        try:install_batch.verify_unchanged();install_batch.require_finalized()
        except InstallEvidenceFailure as error:raise AssertionError(str(error)) from None
    if worker_recovery_batch is not None:
        try:worker_recovery_batch.verify_unchanged();worker_recovery_batch.require_finalized()
        except WorkerRecoveryEvidenceFailure as error:raise AssertionError(str(error)) from None
    if read1161_batch is not None:
        try:read1161_batch.verify_unchanged();read1161_batch.require_finalized()
        except Read1161EvidenceFailure as error:raise AssertionError(str(error)) from None
    if install_precleanup_batch is not None:
        try:install_precleanup_batch.verify_unchanged();install_precleanup_batch.require_finalized()
        except InstallPrecleanupEvidenceFailure as error:raise AssertionError(str(error)) from None
    return {'scope':'ACTUAL_ASSERTION_EVIDENCE_OVER_EXACT_EXISTING94_IDS','database_posture':'D','Brain_acceptance':'PENDING','counts_are_obligations_not_experiments':True,'assertion_status_counts':totals,'case_status_counts':{s:sum(c['status']==s for c in cases) for s in ('PASS','PARTIAL','FAILED','DEPENDENCY_BLOCKED','NOT_RUN')},'selected_cases':94,'executions':ledger['executions'],'cases':cases}

if __name__=='__main__':
    result=build();(REC/'local-f0-qualification-status.json').write_text(json.dumps(result,indent=2)+'\n',encoding='utf8',newline='\n')
    print(json.dumps({k:v for k,v in result.items() if k!='cases'}))
