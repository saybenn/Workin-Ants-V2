def csp_receipt(value,case,success=False):
    need(type(value)is dict and set(value)=={'response_received','shape_valid','acknowledged','exact_output','rollback_acknowledged','diagnostic'},'CSP1_RECEIPT_FIELDS')
    for key in('response_received','shape_valid','exact_output','rollback_acknowledged'):need(type(value[key])is bool,'CSP1_RECEIPT_BOOLEAN')
    need(value['acknowledged']is None or type(value['acknowledged'])is bool,'CSP1_RECEIPT_ACK')
    diag=value['diagnostic']
    if case=='wrong-predecessor':need(diag is None,'CSP1_ZERO_ROWS_NO_DIAGNOSTIC')
    else:
        need(type(diag)is dict and set(diag)=={'matched','sqlstate','exact_primary','source_function','source_body','raise_line','bounded_shape'},'CSP1_DIAGNOSTIC_FIELDS')
        need(diag['sqlstate']in(None,'P0001','42501') and all(type(diag[k])is bool for k in set(diag)-{'sqlstate'}),'CSP1_DIAGNOSTIC_VALUES')
        if success:need(diag['matched']is True and diag['sqlstate']==('P0001'if case=='skipped-revision'else'42501') and all(diag[k]is True for k in set(diag)-{'sqlstate'}),'CSP1_INTENDED_GUARD_REQUIRED')
    if success:need(all(value[k]is True for k in('response_received','shape_valid','acknowledged','exact_output','rollback_acknowledged')),'CSP1_EXPLICIT_ROLLBACK_REQUIRED')
