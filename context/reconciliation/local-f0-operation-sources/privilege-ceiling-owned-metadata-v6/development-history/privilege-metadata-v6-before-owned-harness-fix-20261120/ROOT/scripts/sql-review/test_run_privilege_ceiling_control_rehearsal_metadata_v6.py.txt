    def metadata_fault(self,mode):
        world=World();memory=MemoryEvidence();case=next(c for c in PACKET['cases']if c['name']=='GRANT-wa_app_runtime-DELETE');Runner=M['runner_type'](N,C,P);runner=object.__new__(Runner);runner.native=world.native;runner.evidence=memory;runner.api={'psql':lambda role,cid:['MOCK',role,cid]};runner.cid=CID;runner.failed=False;runner._fingerprinting=False;runner.grants={};runner.source_blobs={M['PACKET']+'/'+case['file']:(ROOT/M['PACKET']/case['file']).read_bytes()};events=[]
        def boundary(label):
            confirmed=not(mode=='live'and label.startswith('AFTER_METADATA_'))
            events.append('closure');memory.value.setdefault('global_client_boundaries',[]).append({'stage':label,'attempted':True,'confirmed':confirmed})
            if not confirmed:raise M['Held']('GLOBAL_CLIENT_CLOSURE_UNCONFIRMED')
        runner.global_closed=boundary
        query,expected,_=M['metadata_query_spec'](case,world.native,P['expected_acl'])
        def run(*args,**kwargs):
            events.append('process-return');self.assertEqual(kwargs['input'],query);self.assertEqual(kwargs['timeout'],60)
            if mode=='throw':raise OSError(CANARY)
            if mode=='timeout':raise subprocess.TimeoutExpired(CANARY,60,output=CANARY,stderr=CANARY)
            if mode in('nonzero','postio'):return types.SimpleNamespace(returncode=3,stdout='',stderr='ERROR:  22023: '+CANARY+'\nDETAIL: '+CANARY+'\n')
            if mode=='boolcode':return types.SimpleNamespace(returncode=False,stdout=json.dumps(expected),stderr='')
            if mode=='badshape':return types.SimpleNamespace(returncode=0,stdout=json.dumps({CANARY:True}),stderr='')
            if mode=='duplicate':return types.SimpleNamespace(returncode=0,stdout='{"x":true,"x":false}',stderr='')
            if mode=='nonfinite':return types.SimpleNamespace(returncode=0,stdout='{"x":NaN}',stderr='')
            if mode=='oversized':return types.SimpleNamespace(returncode=0,stdout=CANARY*20000,stderr='')
            return types.SimpleNamespace(returncode=0,stdout=json.dumps(expected),stderr='')
        save=memory.save
        if mode in('io','postio'):
            def writer():
                receipt=memory.value.get('current',{}).get('metadata_query_receipt')
                if receipt and(mode=='io'or receipt['process_result_received']is True):raise OSError(CANARY)
                save()
            memory.save=writer
        runner.actor=Mock(side_effect=AssertionError('NO_ACTOR_AFTER_METADATA_FAILURE'))
        with patch.object(subprocess,'run',side_effect=run):
            try:runner.negative(case);failure=None
            except BaseException as error:failure=error
        runner.actor.assert_not_called()
        self.assertIsNotNone(failure);self.assertEqual(memory.value['results'],[]);self.assertNotIn(CANARY,json.dumps(memory.value));self.assertIn('closure',events);self.assertNotIn('source_native_privilege_projection',memory.value['current'])if mode not in('live',)else None
        return memory.value,events

    def test_metadata_failure_receipts_and_closure_precede_any_actor(self):
        for mode in('throw','timeout','nonzero','boolcode','badshape','duplicate','nonfinite','oversized','io','postio','live'):
