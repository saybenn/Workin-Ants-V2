class Pipe:
 def __init__(self):self.q=queue.Queue()
 def emit(self,value):self.q.put(value)
 def readline(self,n=-1):return self.q.get()
 def read(self,n=-1):return self.q.get()
 def end(self):self.q.put('')
class Input:
 def __init__(self,process):self.process=process;self.text=''
 def write(self,text):self.text+=text
 def flush(self):text=self.text;self.text='';self.process.execute(text)
 def close(self):self.flush()
class Process:
 def __init__(self,vm,role,c):self.vm=vm;self.model=vm.model(role,c);self.returncode=None;self.stdout=Pipe();self.stderr=Pipe();self.stdin=Input(self);self.ended=False
 def poll(self):return self.returncode
 def wait(self,timeout=None):
  if self.returncode is None:raise subprocess.TimeoutExpired('MOCK_UNENDED',timeout)
  return self.returncode
 def finish(self,code):
  if not self.ended:self.ended=True;self.returncode=code;self.model.closed=True;self.stdout.end();self.stderr.end()
 def terminate(self):self.finish(0)
 def kill(self):self.finish(0)
 def execute(self,text):
  if '\\q' in text:self.model.close();self.finish(self.model.process.returncode);return
  found=re.search(r"\nSELECT '(WA_CTX01_ACK\|[0-9a-f]+)';\n$",text);assert found,'ACTUAL_SESSION_ACK_SQL';sql=text[:found.start()];rows=[];ack=True;errors=''
  if sql.startswith('BEGIN ISOLATION LEVEL '):self.model.begin(re.search(r'BEGIN ISOLATION LEVEL ([A-Z ]+);',sql)[1].lower())
  elif "'login',session_user" in sql and 'pg_backend_pid()::text' in sql:rows=['WA_CTX01_TUPLE|'+json.dumps(self.model.capture())]
  else:rows,ack,errors=self.model.call(sql)
  if self.vm.lost_fixture_ack and 'DISABLE TRIGGER owner_commit' in sql:self.vm.lost_fixture_ack=False;ack=False;self.model.process.returncode=1
  for row in rows:self.stdout.emit(row+'\n')
  if errors:self.stderr.emit(errors)
  if ack:self.stdout.emit(found[1]+'\n')
  else:self.finish(self.model.process.returncode)
