def source_blob(relative,expected=None):
    need(type(relative) is str and not Path(relative).is_absolute() and chr(92) not in relative and all(part not in ('','.','..') for part in relative.split('/')),'RESTORE_SOURCE_PATH')

    raw=path.read_bytes()
    if expected is not None:need(sha(raw)==expected,'RESTORE_SOURCE_ADMISSION_DRIFT')
    return raw
