import datetime
import hashlib
import json

T = Path(__file__).resolve().parent
R = T / 'workinants-correction-pass'
sha = lambda p: hashlib.sha256(p.read_bytes()).hexdigest()
source = T / '1110_results_reviewed_sources_progress_checkpoint_inventory_20261119.json'

path = R / relative
assert sha(path) == inventory['files_sha256'][relative]
raw = path.read_bytes()
board = json.loads(raw)
assert type(board['safety_sql_hold']) is str
assert type(board['owner_resource_hold']) is dict
assert type(board['current_owner_operator']) is dict
board['current_owner_operator']['execution'] = 'COMPLETED_DA5_OWNER7293_FINITE_READ_SUBSET_REVIEWED_EVIDENCE_INTEGRATED_NO_WHOLE_OWNER_API'
board['current_owner_operator']['assertion_integration'] = 'context/reconciliation/local-f0-owner-lease-evidence-adapter-v1-source-validation.json'
board['native_access_source_review'] = 'context/reconciliation/local-f0-native-access-resource-set-v2-source-validation.json'
board['native_access_correction'] = board['native_access_source_review']
board['safety_sql_hold_historical_only'] = True

    handle.write(raw)

inventory['current_action_pointer_clarification'] = {
    'recorded_at_utc': datetime.datetime.now(datetime.timezone.utc).isoformat(),
    'historical_hold_and_Owner_fresh_literal_labels_resolved': True,
    'Native_current_source_review': 'resource-set-v2',

}
output = T / '1110_results_reviewed_sources_progress_checkpoint_inventory_v2_20261119.json'
with output.open('xb') as handle:
    handle.write((json.dumps(inventory, indent=2, allow_nan=False) + '\n').encode())
print(json.dumps({'inventory_sha256': sha(output), 'board_sha256': sha(path)}))
