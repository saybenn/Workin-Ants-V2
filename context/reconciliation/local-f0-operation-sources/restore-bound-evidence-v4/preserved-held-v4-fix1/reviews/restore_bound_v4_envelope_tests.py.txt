class EnvelopeNormalizationTests(unittest.TestCase):
    setUp=BoundEvidenceTests.setUp
    invoke=BoundEvidenceTests.invoke
    proof=BoundEvidenceTests.proof
    safe=BoundEvidenceTests.safe
    def normalize(self,text):
        facts={};value=self.n['normalize_bound_envelope'](text,facts);return value,facts
    def transactions(self,text):
        from pglast import parse_sql,ast as pgast
        return [row.stmt.kind.name for row in parse_sql(text) if isinstance(row.stmt,pgast.TransactionStmt)]
    def test_actual_three_admitted_sources_have_one_final_commit_and_same_business_AST(self):
        from pglast import parse_sql,ast as pgast
        for name,mode in (('source-customer.sql','TERMINAL_COMMIT_RETAINED'),('schedule.sql','TERMINAL_COMMIT_RETAINED'),('app-read.sql','NO_WRAPPER_APPEND_COMMIT')):
            source=self.n['source_text'](name);normalized,facts=self.normalize(source)
            self.assertEqual(self.transactions(normalized),['TRANS_STMT_COMMIT']);self.assertEqual(facts['mode'],mode)
            business=lambda sql:[str(row.stmt) for row in parse_sql(sql) if not isinstance(row.stmt,pgast.TransactionStmt)]
            self.assertEqual(business(source),business(normalized));self.assertTrue(facts['business_bytes_preserved'])
            self.assertEqual(normalized,source if mode=='TERMINAL_COMMIT_RETAINED' else source+'\nCOMMIT;')
    def test_no_wrapper_terminal_and_outer_BEGIN_modes_preserve_literal_business_bytes(self):
        body="-- COMMIT;\nSELECT 'BEGIN; COMMIT; ROLLBACK;', $$COMMIT;$$, \"COMMIT;\";\n"
        for source,mode in ((body,'NO_WRAPPER_APPEND_COMMIT'),(body+'COMMIT;','TERMINAL_COMMIT_RETAINED'),('BEGIN;\n'+body+'COMMIT;','OUTER_BEGIN_REMOVED')):
            normalized,facts=self.normalize(source);self.assertIn(body,normalized);self.assertEqual(facts['mode'],mode);self.assertEqual(facts['emitted_commit_count'],1);self.assertEqual(self.transactions(normalized),['TRANS_STMT_COMMIT'])
    def test_nested_comments_escaped_strings_and_dollar_bodies_are_not_controls(self):
        vectors=["/* BEGIN; /* COMMIT; */ ROLLBACK; */ SELECT 'it''s; COMMIT;';",r"SELECT E'one\'; COMMIT; still string';",'DO $body$ BEGIN PERFORM 1; END; $body$;','SELECT \'-- COMMIT;\', "BEGIN;"; -- ROLLBACK;']
        for source in vectors:
            normalized,facts=self.normalize(source);self.assertTrue(normalized.startswith(source));self.assertEqual(facts['source_begin_count'],0);self.assertEqual(facts['source_commit_count'],0);self.assertEqual(self.transactions(normalized),['TRANS_STMT_COMMIT'])
    def test_unsupported_middle_chained_prepared_rollback_and_ambiguous_envelopes_refuse(self):
        vectors=['BEGIN; SELECT 1;','SELECT 1; COMMIT; COMMIT;','SELECT 1; COMMIT; SELECT 2;','SELECT 1; BEGIN; COMMIT;','BEGIN; BEGIN; SELECT 1; COMMIT;','SELECT 1; ROLLBACK;','SELECT 1; COMMIT AND CHAIN;','BEGIN WORK; SELECT 1; COMMIT;','START TRANSACTION; SELECT 1; COMMIT;','SELECT 1; END;','SELECT 1; ABORT;','SELECT 1; SAVEPOINT a; COMMIT;','SELECT 1; RELEASE a; COMMIT;',"PREPARE TRANSACTION 'x';",'SET TRANSACTION READ ONLY; SELECT 1; COMMIT;','SET SESSION CHARACTERISTICS AS TRANSACTION READ ONLY; SELECT 1; COMMIT;','COMMIT;','SELECT 1','SELECT 1;;','/* unclosed','SELECT $$unclosed;','SELECT \'unclosed;','\\echo unsafe\nSELECT 1;','SELECT 1; commit;','SELECT 1; COMMIT; -- unsupported terminal trivia']
        for source in vectors:
            facts={}
            with self.assertRaises(self.n['Stop']):self.n['normalize_bound_envelope'](source,facts)
            self.assertNotIn(source,json.dumps(facts));self.assertEqual(facts['mode'],'UNCONFIRMED');self.assertFalse(facts['business_bytes_preserved'])
    def test_type_size_control_and_depth_bounds_keep_only_fixed_safe_metadata(self):
        for source in (None,True,[],CANARY+'\x00;',CANARY*40000,'/*'*33+'x'+'*/'*33+'SELECT 1;'):
            facts={}
            with self.assertRaises(self.n['Stop']):self.n['normalize_bound_envelope'](source,facts)
            self.assertNotIn(CANARY,json.dumps(facts))
