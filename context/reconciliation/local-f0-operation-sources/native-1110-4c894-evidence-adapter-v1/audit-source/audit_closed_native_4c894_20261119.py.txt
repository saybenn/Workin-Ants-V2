def need(ok, label):
    if not ok:
        raise AssertionError(label)
    CHECKS[label.split(':')[0]] += 1

def sha(raw): return hashlib.sha256(raw).hexdigest()
def canon(v): return json.dumps(v, sort_keys=True, separators=(',', ':'), ensure_ascii=False).encode()
def digest(v): return sha(canon(v))
def unique(raw):
    def pairs(items):
        result = {}
        for k, v in items:
            need(k not in result, 'json_duplicate_free')
            result[k] = v
        return result
    return json.loads(raw, object_pairs_hook=pairs)

def load(path, expected=None):
    raw = path.read_bytes()
    if expected: need(sha(raw) == expected, 'file_hash:' + path.name)
    return unique(raw)

def hashed(path, expected):
    need(sha(path.read_bytes()) == expected, 'file_hash:' + path.name)

def alltrue(v): return type(v) is dict and bool(v) and all(x is True for x in v.values())
def zero(v): return type(v) is int and v == 0
def aggregate(files): return sha(''.join(p + '\0' + files[p] + '\n' for p in sorted(files)).encode())
def forbidden(*args, **kwargs): raise AssertionError('FORBIDDEN_PROCESS_OR_NETWORK_BOUNDARY')
