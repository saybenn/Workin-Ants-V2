def auth_roles(roles,active):
    for role in roles:
        report['stage']='AUTH_'+role;save();r=auth_call(role)
        if active:
            checks=STATE['auth'].authentication_comparison(r.stdout,role);report.setdefault('auth_comparisons',[]).append({'role':role,'checks':checks});save()
            need(r.returncode==0 and r.stderr==''and all(type(v)is bool and v for v in checks.values()),'AUTH_ACTUAL_SCRAM')

def migrate(operation,label):
    report['stage']=label;save();name=TARGETS[0]['name']
    node_dispatched=False
    def mark_node_dispatch():
        nonlocal node_dispatched
        node_dispatched=True
    def coordinate_process(arguments,**options):
        return source13_process(arguments,_node_dispatch_marker=mark_node_dispatch,**options)
    with _RequiredSafetyBoundary('CLOSE',name,label+'_CLOSED'):
        try:
            result=STATE['host'].coordinate(STATE['nested_raw'],operation,source8_root=MANIFEST['migration_target']['source_snapshot'],source_pins=MANIFEST['packet_files_sha256'],root=ROOT,process=coordinate_process)
        except BaseException as original:
            if node_dispatched:TRANSPORT.unknown=True
            _required_safety_first_exception(original,label)
            if node_dispatched:report['inner_cli_closure']={'node_dispatch_attempted':True,'validated_result_returned':False}
            raise
        if result['status']=='CLI_CLOSURE_UNCONFIRMED_STOP_AND_CLEAN_EXACT_RESOURCE'and result['exit_code']is None:
            # FIRST: latch descendant uncertainty before every may-throw record.
            TRANSPORT.unknown=True
            report['inner_cli_closure']={'reported_unknown':True,'inner_exit_observed_null':True,'outer_callback_returned':True}
            _required_safety_first_exception(Stop('REPORTED_INNER_CLI_CLOSURE_UNCONFIRMED'),label)
            need(False,'REPORTED_INNER_CLI_CLOSURE_UNCONFIRMED')
        report.setdefault('migration_results',[]).append(result);save()
        need(STATE['host'].runner_succeeded(result),'GENUINE_PRISMA_OPERATION_FAILED')
    return result
