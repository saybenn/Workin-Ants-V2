class Stop(Exception):
    def __init__(self,code,details=None):self.code=code;self.details=details or {};self.stage=report.get('stage','ENTRY');super().__init__(code)

def need(ok,code,details=None):
    if not ok:raise Stop(code,details)

def _remember_first_save_io(phase,error):
    global FIRST_SAVE_IO
    if FIRST_SAVE_IO is not None:return
    try:
        category=next((name for kind,name in ((PermissionError,'PermissionError'),(FileNotFoundError,'FileNotFoundError'),(UnicodeError,'UnicodeError'),(OSError,'OSError'),(TypeError,'TypeError'),(ValueError,'ValueError')) if isinstance(error,kind)),'OTHER')
        try:number=getattr(error,'winerror',None)
        except BaseException:number=None
        value={'phase':phase,'exception_class':category,'winerror':number if type(number) is int and number in (5,32,33) else None};FIRST_SAVE_IO=value
        try:report['first_save_io']=dict(value)
        except BaseException:pass
    except BaseException:pass

def save():
    stream=None;first=None;phase='TEMP_OPEN';temp=OUT/'run-evidence.json.tmp'
    try:
        stream=temp.open('w',encoding='utf8',newline='\n')
        phase='SERIALIZE_REDACT';text=json.dumps(report,indent=2)+'\n'
        for secret in all_secrets:text=text.replace(secret,'[REDACTED]')
        phase='TEMP_WRITE';stream.write(text)
        phase='TEMP_FLUSH';stream.flush()
        phase='TEMP_FSYNC';os.fsync(stream.fileno())
    except BaseException as error:first=error;_remember_first_save_io(phase,error)
    finally:
        if stream is not None:
            try:stream.close()
            except BaseException as error:
                _remember_first_save_io('TEMP_CLOSE',error)
                if first is None:first=error
    if first is not None:mark_evidence_incomplete();raise first
    try:temp.replace(OUT/'run-evidence.json')
    except BaseException as error:_remember_first_save_io('ATOMIC_REPLACE',error);mark_evidence_incomplete();raise

def finish_report(success_status):
    def acceptable():
        return evidence_is_complete() and not report['unexpected'] and report['status']==success_status
    if not acceptable():report['status']='STOPPED_ATTEMPT_PRESERVED';report.pop('qualified_native_snapshot_sha256',None)
    report['finished_at']=datetime.datetime.now(datetime.timezone.utc).isoformat()
    best_effort_save()
    if not acceptable():report['status']='STOPPED_ATTEMPT_PRESERVED';report.pop('qualified_native_snapshot_sha256',None)
    try:
        print(json.dumps({'status':report['status'],'evidence_complete':evidence_is_complete(),'persistence_failure':report.get('persistence_failure',False),'first_save_io':FIRST_SAVE_IO,'evidence':str(OUT/'run-evidence.json')}),flush=True)
    except BaseException:
        mark_evidence_incomplete('EVIDENCE_OUTPUT_UNCONFIRMED');best_effort_save()
    return 0 if acceptable() else 1


def event(stage,status,**safe):
    try:
        report['stage']=stage;report['events'].append({'stage':stage,'status':status,**safe});
        mapped={'LOCAL_PREFLIGHT':'LOCAL','RESOURCE_READY':'RESOURCES','CONTROLLED_CLEANUP':'CLEANUP'}.get(stage,stage)
        if mapped in report.get('steps',{}):report['steps'][mapped]=status
        save()
        print(json.dumps({'stage':stage,'status':status,**safe}),flush=True)
    except BaseException:
        mark_evidence_incomplete('EVIDENCE_EVENT_UNCONFIRMED');raise

def safe_error(stderr):
    found=[]
    for line in stderr.splitlines():
        m=re.fullmatch(r'(?:psql:(?:<stdin>|-):([0-9]+): )?(?:ERROR|FATAL):\s+([0-9A-Z]{5}):\s+(.*)',line)
        if not m:continue
        state=m[2];message=m[3];diagnostic='UNLISTED';obj=None
        known={'division by zero':'DIVISION_BY_ZERO','foreign Customer principal':'CUSTOMER_PRINCIPAL_GUARD','missing durable default-Free intent':'CUSTOMER_DEFAULT_FREE_GUARD','cleanup requires original bootstrap grantor':'WRONG_FINALIZATION_GRANTOR','temporary installation membership remains':'TEMPORARY_MEMBERSHIP_REMAINS','temporary function creation authority remains':'TEMPORARY_CREATE_REMAINS'}
        if message in known:diagnostic=known[message]
        found.append({'sqlstate':state,'diagnostic':diagnostic,'source_line':int(m[1]) if m[1] else None,**({'object':obj} if obj else {})})
    return found

def sql(target,role,text,kind,timeout=120,expected_error=None,installation=False):
    report['stage']=kind;save()
    args=psql_args(target,role)
    if installation:
        args+=['-c',"SELECT jsonb_build_object('session_user',session_user,'current_user',current_user,'system_user',system_user,'database',current_database());",'-f','-']
    result=command(args,text,timeout)
    details={'target':target,'role':role,'operation':kind,'exit_code':result.returncode,'errors':safe_error(result.stderr),'client_exited':True}
    if kind!='CREDENTIAL_PROVISION':details['input_sha256']=hashlib.sha256(text.encode()).hexdigest()
    if installation and result.stdout.splitlines():
        try:
            identity=json.loads(result.stdout.splitlines()[0])
            if set(identity)=={'session_user','current_user','system_user','database'}:details['actual_install_connection']=identity
        except json.JSONDecodeError:pass
    report['events'].append(details);save()
    if expected_error:
        need(result.returncode==3 and len(details['errors'])==1 and details['errors'][0]['sqlstate']==expected_error and details['errors'][0]['diagnostic']=='DIVISION_BY_ZERO','EXPECTED_INSTALL_FAILURE_NOT_PROVEN',details)
    else:need(result.returncode==0,'SQL_OPERATION_FAILED',details)
    return result.stdout.strip(),details

def observe(target,text,kind):
    stdout,details=sql(target,'postgres',text,kind)
    return json.loads(stdout)

def _required_safety_text(selector):
    allowed={'GLOBAL_BACKENDS','FINALIZATION_BACKENDS','FINALIZATION_RESULT','ORIGINAL_GRANTOR_FINALIZATION'}
    if _SAFETY_SQL_PROFILE=='INSTALL':allowed|={'INSTALL_BOOTSTRAP_ELIGIBILITY','INSTALL_ORIGINAL_CLEANUP_RESULT'}
    need(_SAFETY_SQL_PROFILE in ('STANDARD','INSTALL') and selector in allowed,'REQUIRED_SAFETY_SELECTOR_NOT_ADMITTED')
    if selector!='ORIGINAL_GRANTOR_FINALIZATION':return _SAFETY_SQL_RECIPES[selector]
    try:
        raw_path=ROOT/_SAFETY_FINALIZER;metadata=raw_path.lstat()
        need(_required_safety_stat.S_ISREG(metadata.st_mode) and not raw_path.is_symlink(),'REQUIRED_FINALIZER_SOURCE_UNCONFIRMED')
        raw=read_pinned(_SAFETY_FINALIZER)
    except BaseException:raise Stop('REQUIRED_FINALIZER_SOURCE_UNCONFIRMED') from None
    need(type(raw) is bytes and hashlib.sha256(raw).hexdigest()==_SAFETY_FINALIZER_SHA,'REQUIRED_FINALIZER_SOURCE_UNCONFIRMED')
    try:return raw.decode('utf8')
    except UnicodeError:raise Stop('REQUIRED_FINALIZER_SOURCE_UNCONFIRMED') from None


def _required_safety_decode(raw,selector):
    need(type(raw) is str,'REQUIRED_SAFETY_RESPONSE_UNCONFIRMED')
    try:bounded=len(raw.encode('utf8'))<=65536
    except UnicodeError:bounded=False
    need(bounded,'REQUIRED_SAFETY_RESPONSE_UNCONFIRMED')
    def pairs(items):
        row={}
        for key,value in items:
            need(key not in row,'REQUIRED_SAFETY_RESPONSE_UNCONFIRMED');row[key]=value
        return row
    def bad_constant(_):raise Stop('REQUIRED_SAFETY_RESPONSE_UNCONFIRMED')
    try:value=json.loads(raw,object_pairs_hook=pairs,parse_constant=bad_constant)
    except BaseException:raise Stop('REQUIRED_SAFETY_RESPONSE_UNCONFIRMED') from None
    need(type(value) is dict,'REQUIRED_SAFETY_RESPONSE_UNCONFIRMED')
    if selector in ('GLOBAL_BACKENDS','FINALIZATION_BACKENDS'):
        need(set(value)=={'migrator','other_client_backends'} and all(type(value[k]) is int and value[k]>=0 for k in value),'REQUIRED_SAFETY_RESPONSE_UNCONFIRMED')
    elif selector=='FINALIZATION_RESULT':
        need(set(value)=={'grantor','auth','remaining_memberships','leaf_create','active_install_connections'} and value['grantor']=='postgres' and value['auth']=='scram-sha-256:postgres' and type(value['leaf_create']) is dict and set(value['leaf_create'])==set(LEAVES) and all(type(v) is bool for v in value['leaf_create'].values()) and all(type(value[k]) is int and value[k]>=0 for k in ('remaining_memberships','active_install_connections')),'REQUIRED_SAFETY_RESPONSE_UNCONFIRMED')
    elif selector=='INSTALL_BOOTSTRAP_ELIGIBILITY':
        need(set(value)=={'source_roles','login','auth'} and type(value['source_roles']) is int and value['source_roles']>=0 and value['login']=='postgres' and value['auth']=='scram-sha-256:postgres','REQUIRED_SAFETY_RESPONSE_UNCONFIRMED')
    elif selector=='INSTALL_ORIGINAL_CLEANUP_RESULT':
        need(set(value)=={'temporary_memberships','leaf_create','auth','grantor'} and type(value['temporary_memberships']) is int and value['temporary_memberships']>=0 and type(value['leaf_create']) is bool and value['auth']=='scram-sha-256:postgres' and value['grantor']=='postgres','REQUIRED_SAFETY_RESPONSE_UNCONFIRMED')
    return value


def _required_safety_sql(target,selector):
    need(type(selector) is str and selector in _SAFETY_SQL_LABELS,'REQUIRED_SAFETY_SELECTOR_NOT_ADMITTED')
    need(type(target) is str and target in created_containers and target in RESOURCE_CONTAINERS and type(created_containers[target]) is str and re.fullmatch('[0-9a-f]{64}',created_containers[target]),'REQUIRED_SAFETY_OWNED_TARGET')
    report['stage']=_SAFETY_SQL_LABELS[selector]
    text=_required_safety_text(selector)
    args=psql_args(target,'postgres')
    need(type(args) is list and args.count('-U')==1 and args[args.index('-U')+1]=='postgres' and created_containers[target] in args,'REQUIRED_SAFETY_POSTGRES_TRANSPORT')
    stage=_SAFETY_SQL_LABELS[selector];report['stage']=stage;_required_safety_save();_required_safety_event('REQUIRED_SAFETY_SQL','ATTEMPTED',target=target,operation=selector);report['stage']=stage
    details={'target':target,'role':'postgres','operation':stage,'input_sha256':hashlib.sha256(text.encode()).hexdigest(),'client_exited':False,'exit_code':None,'response_fields_valid':False}
    try:
        result=command(args,text,120)
        code=getattr(result,'returncode',None);stdout=getattr(result,'stdout',None);stderr=getattr(result,'stderr',None)
        details['client_exited']=type(code) is int and 0<=code<=255;details['exit_code']=code if details['client_exited'] else None
        need(details['client_exited'] and code==0 and type(stdout) is str and type(stderr) is str and stderr=='','REQUIRED_SAFETY_COMMAND_UNCONFIRMED')
        if selector=='ORIGINAL_GRANTOR_FINALIZATION':
            need(stdout.strip()=='','REQUIRED_SAFETY_RESPONSE_UNCONFIRMED');value=None
        else:value=_required_safety_decode(stdout,selector)
        details['response_fields_valid']=True;return value
    finally:
        report.setdefault('events',[]).append(details);_required_safety_save()

def finalize(name):
    need(not TRANSPORT.unknown,'FINALIZATION_UNKNOWN_TRANSPORT')
    if name in finalization_attempted:return
    finalization_attempted.add(name)
    active=_required_safety_sql(name,'FINALIZATION_BACKENDS')
    valid_active=(type(active) is dict and set(active)=={'migrator','other_client_backends'} and all(type(active[key]) is int and active[key]>=0 for key in ('migrator','other_client_backends')))
    if valid_active:report['targets'][name]['connections_before_finalization']=active;_required_safety_save()
    need(valid_active and all(active[key]==0 for key in ('migrator','other_client_backends')),'FINALIZATION_BLOCKED_ACTIVE_OPERATION',{'target':name})
    first=None;observation=None
    try:_required_safety_sql(name,'ORIGINAL_GRANTOR_FINALIZATION')
    except BaseException as error:first=error;_required_safety_first_exception(error)
    finally:
        try:observation=_required_safety_sql(name,'FINALIZATION_RESULT')
        except BaseException:
            _required_safety_failure('FINALIZATION_POST_OBSERVER','FINALIZATION_POST_OBSERVER_UNCONFIRMED')
            if first is None:raise
    valid_observation=(type(observation) is dict and set(observation)=={'grantor','auth','remaining_memberships','leaf_create','active_install_connections'} and observation['grantor']=='postgres' and observation['auth']=='scram-sha-256:postgres' and type(observation['leaf_create']) is dict and set(observation['leaf_create'])==set(LEAVES) and all(type(value) is bool for value in observation['leaf_create'].values()) and all(type(observation[key]) is int and observation[key]>=0 for key in ('remaining_memberships','active_install_connections')))
    if valid_observation:
        report['targets'][name]['finalization_observation']=observation;report['targets'][name]['finalization_observation_status']='OBSERVED_NOT_QUALIFIED';_required_safety_save()
    if first is not None:
        report['targets'][name]['finalization']='UNCONFIRMED';_required_safety_save();raise first
    need(valid_observation and observation['grantor']=='postgres' and observation['auth']=='scram-sha-256:postgres' and observation['remaining_memberships']==0 and observation['active_install_connections']==0 and not any(observation['leaf_create'].values()),'FINALIZATION_VERIFICATION_FAILED',{'target':name})
    finalized.add(name);report['targets'][name]['finalization']='PASS';_required_safety_event('FINALIZATION','PASS',target=name)
    if not _required_safety_evidence_complete():report['targets'][name]['finalization']='OBSERVED_CLEANUP_EVIDENCE_INCOMPLETE';_required_safety_save()
    _required_safety_barrier()


def secret_scan():
    scanned=0
    for path in OUT.rglob('*'):
        if not path.is_file() or PRIVATE in path.parents:continue
        text=path.read_text(encoding='utf8');scanned+=1
        for value in all_secrets:
            need(value not in text,'SECRET_EVIDENCE_SCAN_FAILED',{'file':str(path.relative_to(OUT))})
    report['sanitization']={'files_scanned':scanned,'actual_password_values_found':False,'unrestricted_subprocess_output_persisted':False,'raw_context_handles_persisted':False,'coordinator_evidence_uses_reviewed_allowlist':True};save()

def cleanup():
    errors=[]
    def attempt(label,target,action):
        try:action()
        except BaseException as exc:
            item={'stage':'CLEANUP','target':target,'code':exc.code if isinstance(exc,Stop) else label}
            errors.append(item);report['cleanup'][target]={'status':'BLOCKED','failure':item['code']};best_effort_save()
    def container_cleanup(name):
        value=inspect_container(name)
        if name not in created_containers:
            if value is None or not owned(value['Config']['Labels']):return
            created_containers[name]=value['Id']
        if value is None:
            report['cleanup'][name]={'status':'ABSENT_AT_CLEANUP'};best_effort_save();return
        identity=created_containers[name]
        need(value.get('Name')=='/'+name and value['Id']==identity and owned(value['Config']['Labels']),'CLEANUP_CONTAINER_OWNERSHIP_MISMATCH',{'target':name})
        need(docker(['container','rm','-f',identity]).returncode==0,'EXACT_CONTAINER_REMOVAL_FAILED',{'target':name})
        report['cleanup'][name]={'removed':True,'verified_id':identity,'name_and_both_labels_verified':True};best_effort_save()
    def volume_cleanup(name):
        value=inspect_volume(name)
        if name not in created_volumes:
            if value is None or not owned(value.get('Labels') or {}):return
            created_volumes[name]=value['CreatedAt']
        if value is None:
            report['cleanup'][name]={'status':'ABSENT_AT_CLEANUP'};best_effort_save();return
        need(value.get('Name')==name and value['CreatedAt']==created_volumes[name] and owned(value.get('Labels') or {}),'CLEANUP_VOLUME_OWNERSHIP_MISMATCH',{'target':name})
        need(docker(['volume','rm',name]).returncode==0,'EXACT_VOLUME_REMOVAL_FAILED',{'target':name})
        report['cleanup'][name]={'removed':True,'name_labels_creation_identity_verified':True};best_effort_save()
    try:
        for name in sorted(attempted_containers):attempt('CONTAINER_CLEANUP_FAILED',name,lambda name=name:container_cleanup(name))
        for name in sorted(attempted_volumes):attempt('VOLUME_CLEANUP_FAILED',name,lambda name=name:volume_cleanup(name))
    finally:
        deleted=[]
        for path in secret_paths:
            def delete_secret(path=path):
                private_resolved=PRIVATE.resolve()
                need(private_resolved==OUT.resolve()/'private' and private_resolved.is_relative_to(TASK.resolve()) and not private_resolved.is_relative_to(ROOT.resolve()),'SECRET_CLEANUP_DIRECTORY_MISMATCH')
                need(path.resolve().parent==private_resolved and path.name == 'ok-bootstrap-password','SECRET_CLEANUP_PATH_MISMATCH')
                if not confirmed_path_absent(path):path.unlink()
                deleted.append(path.name)
            attempt('SECRET_FILE_REMOVAL_FAILED',path.name,delete_secret)
        attempt('PRIVATE_DIRECTORY_REMOVAL_FAILED','private_directory',lambda: None if confirmed_path_absent(PRIVATE) else PRIVATE.rmdir())
        report['host_secret_cleanup']={'deleted':deleted,'private_directory_absent':False}
        try:report['host_secret_cleanup']['private_directory_absent']=confirmed_path_absent(PRIVATE)
        except BaseException:errors.append({'stage':'CLEANUP','code':'PRIVATE_ABSENCE_OBSERVATION_FAILED'})
        best_effort_save()
    report['cleanup_summary']={'resource_absence_observation_completed':False,'resource_presence_after':None,'private_directory_absent':False}
    try:
        containers,volumes=inventory()
        presence={name:name in containers for name in RESOURCE_CONTAINERS}|{name:name in volumes for name in RESOURCE_VOLUMES}
        report['cleanup_summary']={'resource_absence_observation_completed':True,'resource_presence_after':presence,'test_owned_secret_files_deleted':report['host_secret_cleanup']['deleted'],'private_directory_absent':report['host_secret_cleanup']['private_directory_absent'],'image_removed':False,'unrelated_resources_changed':False}
        need(not any(presence.values()),'RESOURCE_ABSENCE_NOT_CONFIRMED')
    except BaseException as exc:
        errors.append({'stage':'CLEANUP','code':exc.code if isinstance(exc,Stop) else 'CLEANUP_ABSENCE_OBSERVATION_FAILED'})
    if errors or not evidence_is_complete():
        report['unexpected'].extend(errors);best_effort_save();raise Stop('CONTROLLED_CLEANUP_INCOMPLETE')
    if not best_effort_event('CONTROLLED_CLEANUP','PASS',all_created_names_absent=True,expected_resource_names=len(RESOURCE_CONTAINERS)+len(RESOURCE_VOLUMES),test_secret_files_absent=True):raise Stop('CLEANUP_EVIDENCE_UNCONFIRMED')
    report['steps']['TWO-HISTORY-CLEANUP']='PASS'


# NEW owning-only earliest failure capture. No old helper/module is patched.
def _safe_failure_stage(value):
    return value if type(value)is str and value in SAFE_FAILURE_STAGES else 'UNEXPECTED_STAGE'

def _safe_failure_code(value):
    return value if type(value)is str and value in SAFE_FAILURE_CODES else 'UNEXPECTED_OWNED_FAILURE'

def _latch_first_failure(stage,code):
    global FIRST_OPERATION_FAILURE
    safe={'stage':_safe_failure_stage(stage),'code':_safe_failure_code(code)}
    if FIRST_OPERATION_FAILURE is None:FIRST_OPERATION_FAILURE=dict(safe)
    # The in-memory first cause is established before any persistence attempt.
    try:report['primary_failure']=dict(FIRST_OPERATION_FAILURE)
    except BaseException:pass
    return safe

def record_failure(stage,code,primary=False):
    safe=_latch_first_failure(stage,code)
    try:
        report['unexpected'].append(safe);report['status']='STOPPED_ATTEMPT_PRESERVED'
        report.pop('qualified_native_snapshot_sha256',None)
    except BaseException:mark_evidence_incomplete()
    return best_effort_save()

def _required_safety_first_exception(error,stage=None):
    """Never throw or retain arbitrary exception messages/probe dictionaries."""
    try:
        candidate=getattr(error,'code',None)
        if not(type(candidate)is str and candidate in SAFE_FAILURE_CODES):
            args=getattr(error,'args',())
            candidate=args[0]if type(args)is tuple and len(args)==1 and type(args[0])is str and args[0]in SAFE_FAILURE_CODES else'UNEXPECTED_OWNED_FAILURE'
        original_stage=stage if stage is not None else getattr(error,'stage',report.get('stage','ENTRY'))
        record_failure(original_stage,candidate)
    except BaseException:
        # A failed secondary record may never erase the already latched cause.
        try:_latch_first_failure(stage,'UNEXPECTED_OWNED_FAILURE')
        except BaseException:pass

class _RequiredSafetyBoundary:
    """Latch every originating exception before running eligible safety work."""
    def __init__(self,action,target,stage=None):
        need(action in('CLOSE','FINALIZE'),'REQUIRED_SAFETY_BOUNDARY_ACTION')
        self.action=action;self.target=target;self.stage=stage
        self.origin_stage=_safe_failure_stage(report.get('stage','ENTRY'))
    def __enter__(self):return self
    def __exit__(self,kind,original,traceback):
        if original is not None:_required_safety_first_exception(original,self.origin_stage)
        try:
            if self.action=='CLOSE':closed_backends(self.target,self.stage)
            else:finalize(self.target)
        except BaseException as secondary:
            _required_safety_first_exception(secondary,report.get('stage','ENTRY'))
            _required_safety_failure('REQUIRED_SAFETY_COMPOSITION','REQUIRED_SAFETY_ACTION_UNCONFIRMED')
            if original is None:raise
        # Following semantic checks still belong to the completed operation,
        # rather than to the successful close observer's temporary phase.
        report['stage']=self.origin_stage
        return False

def hold(ok,code):

def strict_json(raw):
    hold(type(raw)in(bytes,str) and 0<len(raw)<=32*1024*1024,'BOUNDED_JSON')
    def pairs(items):
        out={}
        for k,v in items:hold(k not in out,'DUPLICATE_JSON');out[k]=v
        return out
    try:value=json.loads(raw,object_pairs_hook=pairs,parse_constant=lambda _:(_ for _ in ()).throw(TransportStop('NONFINITE_JSON')))
    except (ValueError,UnicodeError,RecursionError):raise TransportStop('MALFORMED_JSON')from None
    def typed(v,d=0):
        hold(d<=48,'JSON_DEPTH')

            for x in v.values():typed(x,d+1)

def closed_bytes(path,cap=32*1024*1024,origin=None):
    """Two closed reads; ctime comparisons use the same representation."""
    path=Path(path)
    hold(path.is_absolute() and not any(p.is_symlink()for p in(path,*path.parents)),'CLOSED_PATH')
    def key(s):return(s.st_dev,s.st_ino,s.st_mode,s.st_nlink,s.st_size,s.st_mtime_ns)
    def once(token=None):
        a=path.lstat();attrs=getattr(a,'st_file_attributes',0)
        hold(stat.S_ISREG(a.st_mode)and a.st_nlink==1 and 0<=a.st_size<=cap and not attrs&getattr(stat,'FILE_ATTRIBUTE_REPARSE_POINT',1024),'CLOSED_REGULAR_FILE')
        hold(token is None or(key(a),a.st_ctime_ns)==token,'CLOSED_IDENTITY')
        with path.open('rb')as stream:
            h=os.fstat(stream.fileno());hold(key(h)==key(a),'CLOSED_DESCRIPTOR')
            raw=stream.read(cap+1);end=os.fstat(stream.fileno())
            hold(key(h)==key(end)and h.st_ctime_ns==end.st_ctime_ns,'CLOSED_DESCRIPTOR_DRIFT')
        b=path.lstat();hold(key(a)==key(b)and a.st_ctime_ns==b.st_ctime_ns and len(raw)==a.st_size,'CLOSED_PATH_DRIFT')
        return raw,(key(b),b.st_ctime_ns)
    raw,token=once(None if origin is None else origin[2]);again,token2=once(token)
    hold(raw==again and token==token2,'CLOSED_BYTES_DRIFT')
    result=(digest_bytes(raw),len(raw),token)
    hold(origin is None or result==origin,'CLOSED_ORIGIN_DRIFT');return raw,result

class GitBatch:
    entries:tuple
    commit:str
    @property
    def request(self):return ''.join(self.commit+':'+name+'\n'for name,raw in self.entries).encode('utf8')
    @property
    def cap(self):return sum(len(raw)+48+len(str(len(raw)))for _,raw in self.entries)

def git_batches(blobs,commit):

        size=len(raw)+48+len(str(len(raw)));hold(size<=16777216,'GIT_SINGLE_BLOB_TOO_LARGE')
        if items and total+size>16777216:groups.append(GitBatch(tuple(items),commit));items=[];total=0
        items.append((name,raw));total+=size
    if items:groups.append(GitBatch(tuple(items),commit))

def verify_git_batch(batch,raw):
    hold(type(batch)is GitBatch and type(raw)is bytes and len(raw)==batch.cap,'GIT_COMPLETE_SIZE')
    at=0
    for name,body in batch.entries:
        end=raw.find(b'\n',at);hold(end>=at,'GIT_HEADER')
        oid=hashlib.sha1(b'blob '+str(len(body)).encode()+b'\0'+body).hexdigest().encode()
        hold(raw[at:end]==oid+b' blob '+str(len(body)).encode(),'GIT_HEADER');at=end+1
        hold(raw[at:at+len(body)]==body and raw[at+len(body):at+len(body)+1]==b'\n','GIT_COMMITTED_BYTES');at+=len(body)+1
    hold(at==len(raw),'GIT_TRAILING_BYTES');return True

class Transport:

        try:
            hold(all(len(a.encode('utf8'))<=8192 for a in args)and sum(len(a.encode('utf8'))for a in args)<=16384 and len(subprocess.list2cmdline(args).encode('utf-16-le'))//2<=30000,'PROCESS_ARGUMENT_BOUNDS')

        thread_errors=[];error_lock=threading.Lock();cleanup_errors=[];cleanup_threads=[]
        def remember(code):

            bad.set()
        def check_work():
            if bad.is_set()and thread_errors[0][1]<=deadline:raise TransportStop(thread_errors[0][0])
            hold(clock()<deadline,'PROCESS_DEADLINE_EXCEEDED')
        def remaining(end):return max(0,end-clock())
        try:

            p=self.io.Popen(args,stdin=subprocess.PIPE,stdout=subprocess.PIPE,stderr=subprocess.PIPE,cwd=str(self.cwd),shell=False)
            check_work() # Explicit approved post-spawn check before payload delivery.
            def reader(stream,index):
                try:
                    while True:
                        chunk=stream.read(4096)
                        if not chunk:break
                        if type(chunk)is not bytes or len(buffers[index])+len(chunk)>limits[index]:remember('PROCESS_BOUNDED_COMPLETE_OUTPUT');break
                        buffers[index].extend(chunk)
                except BaseException:remember('PROCESS_BOUNDED_COMPLETE_OUTPUT')
            def writer():
                try:
                    if payload:
                        written=p.stdin.write(payload);hold(type(written)is int and written==len(payload),'PROCESS_STDIN_WRITE_UNCONFIRMED')
                except BaseException:remember('PROCESS_STDIN_WRITE_UNCONFIRMED')
                finally:
                    try:p.stdin.close()
                    except BaseException:remember('PROCESS_STDIN_CLOSE_UNCONFIRMED')
            for i,s in enumerate((p.stdout,p.stderr)):
                try:th=self.io.Thread(target=reader,args=(s,i),daemon=True);threads.append(th);th.start()
                except BaseException:raise TransportStop('PROCESS_THREAD_START_UNCONFIRMED')from None
            try:th=self.io.Thread(target=writer,daemon=True);threads.append(th);th.start()
            except BaseException:raise TransportStop('PROCESS_THREAD_START_UNCONFIRMED')from None
            while True:
                check_work()
                try:code=p.wait(timeout=min(.05,remaining(deadline)));break
                except subprocess.TimeoutExpired:continue
            hold(type(code)is int,'PROCESS_EXIT_TYPE')
            for th in threads:
                while th.is_alive():
                    check_work()
                    try:th.join(timeout=min(.05,remaining(deadline)))
                    except BaseException:raise TransportStop('PROCESS_THREAD_JOIN_UNCONFIRMED')from None
            check_work()
        except BaseException as error:failure=error
        finally:
            def observed_exit():
                try:return p is not None and type(p.poll())is int
                except BaseException:return False
            closed=False;alive=True
            if p is not None:
                cleanup_deadline=clock()+2
                def cleanup_call(action,label):
                    def run():
                        try:action()
                        except BaseException:cleanup_errors.append(label)
                    try:th=self.io.Thread(target=run,daemon=True);cleanup_threads.append(th);th.start();return th
                    except BaseException:cleanup_errors.append(label);return None
                def cleanup_join(th,limit=None):
                    if th is None:return
                    try:th.join(timeout=remaining(cleanup_deadline)if limit is None else min(limit,remaining(cleanup_deadline)))
                    except BaseException:cleanup_errors.append('PROCESS_CLOSE_UNCONFIRMED')
                if not observed_exit():

                    if th is not None and not th.is_alive():
                        try:p.wait(timeout=min(.25,remaining(cleanup_deadline)))
                        except BaseException:pass
                    if not observed_exit():

                        if th is not None and not th.is_alive():
                            try:p.wait(timeout=remaining(cleanup_deadline))
                            except BaseException:pass
                for stream in(p.stdin,p.stdout,p.stderr):cleanup_call(stream.close,'PROCESS_PIPE_CLOSE_UNCONFIRMED')
                for th in(*threads,*cleanup_threads):cleanup_join(th)
                try:closed=all(s.closed is True for s in(p.stdin,p.stdout,p.stderr));alive=any(th.is_alive()for th in(*threads,*cleanup_threads))
                except BaseException:pass
                if failure is None:
                    if not observed_exit():failure=TransportStop('PROCESS_EXIT_UNCONFIRMED')
                    elif alive:failure=TransportStop('PROCESS_CLOSE_UNCONFIRMED')
                    elif cleanup_errors or not closed:failure=TransportStop('PROCESS_PIPE_CLOSE_UNCONFIRMED')
                    elif bad.is_set():failure=TransportStop(thread_errors[0][0])
            confirmed=observed_exit()and closed and not alive and not cleanup_errors
            normal_binary_result=failure is None and confirmed
            remote_unconfirmed=remote_exec and attempted and not normal_binary_result
            self.unknown|=not confirmed or remote_unconfirmed
            observation={'profile':profile,'stdin_bytes':0 if payload is None else len(payload),'stdout_bytes':len(buffers[0]),'stderr_bytes':len(buffers[1]),'process_exit_confirmed':observed_exit(),'pipes_closed':closed,'threads_closed':not alive,'closure_confirmed':confirmed,'normal_binary_result_returned':normal_binary_result,'remote_exec_attempted':remote_exec and attempted,'remote_exit_unconfirmed':remote_unconfirmed,'failed':failure is not None}
            self.observations.append(observation)
            if failure is not None:
                if not isinstance(failure,TransportStop):failure=TransportStop('PROCESS_BOUNDARY_UNEXPECTED')
                failure.closure_confirmed=confirmed;raise failure from None
        return Result(code,bytes(buffers[0]),bytes(buffers[1]))
    def text(self,args,input=None,timeout=30,profile='control'):
        hold(input is None or type(input)is str,'PROCESS_TEXT_INPUT')
        try:raw=None if input is None else input.encode('utf8',errors='strict')
        except UnicodeError:raise TransportStop('PROCESS_TEXT_ENCODING')from None

        try:return Result(result.returncode,result.stdout.decode('utf8',errors='strict'),result.stderr.decode('utf8',errors='strict'))
