class QualificationBatch:
    """Immutable private capture, sticky refusal, complete final byte barrier."""
    def __init__(self,root,_single=None):

        if _single is not None:need(_single in CLOSED_CLAIMS,'UNSUPPORTED_CLAIM');self.__expected=frozenset((_single,))
        self.__seen=set();self.__failed=False;self.__finalized=False
    def validate_claim(self,case_id,assertion_id,status,execution,root=None,json_pointer=None,ledger_row=None):
        try:

            need(key in self.__expected and key not in self.__seen,'UNSUPPORTED_OR_DUPLICATE_CLAIM')

            need(canonical(execution)==self.__execution,'EXACT_DERIVED_RECORD');need(canonical(ledger_row)==self.__rows[key],'EXACT_NEW_ROW_AND_PRIOR_HISTORY')

    def verify_unchanged(self):
