def bounded_component_reference(fixed,reference):
    """Only one closed, regular, single-link byte sequence enters JSON parsing."""
    limit=16*1024*1024
    try:
        metadata=fixed.lstat();reference['present']=True
        need(type(metadata.st_mode)is int and stat.S_ISREG(metadata.st_mode)and not fixed.is_symlink()and type(metadata.st_nlink)is int and metadata.st_nlink==1 and type(metadata.st_size)is int and 0<=metadata.st_size<=limit,'SCOPED_REFERENCE_SHAPE_OR_LIMIT')
        with fixed.open('rb')as stream:
            opened=os.fstat(stream.fileno());need((opened.st_dev,opened.st_ino,opened.st_size,opened.st_mtime_ns)==(metadata.st_dev,metadata.st_ino,metadata.st_size,metadata.st_mtime_ns)and stat.S_ISREG(opened.st_mode)and opened.st_nlink==1,'SCOPED_REFERENCE_OPEN_IDENTITY')
            raw=stream.read(limit+1);after=os.fstat(stream.fileno())
            need(type(raw)is bytes and len(raw)==metadata.st_size and len(raw)<=limit and (after.st_dev,after.st_ino,after.st_size,after.st_mtime_ns,after.st_nlink)==(opened.st_dev,opened.st_ino,opened.st_size,opened.st_mtime_ns,1),'SCOPED_REFERENCE_GROWTH_OR_IDENTITY')
        closed=fixed.lstat();need(not fixed.is_symlink()and (closed.st_dev,closed.st_ino,closed.st_size,closed.st_mtime_ns,closed.st_nlink)==(after.st_dev,after.st_ino,after.st_size,after.st_mtime_ns,1),'SCOPED_REFERENCE_CLOSED_IDENTITY')
        reference.update(read_status='CLOSED_BOUNDED_BYTES',bytes=len(raw),sha256=hashlib.sha256(raw).hexdigest())
        return unique_json(raw)
    except FileNotFoundError:reference.update(present=False,read_status='ABSENT');raise Stop('SCOPED_COMPONENT_REFERENCE_ABSENT')from None
    except BaseException:
        reference['read_status']='UNAVAILABLE_OR_REJECTED';raise Stop('SCOPED_COMPONENT_REFERENCE_UNCONFIRMED')from None
