def source_uuid_namespace(value,found=None,depth=0):
    found=set() if found is None else found
    need(depth<=40,'CONTROL_NAMESPACE_DEPTH')
    if type(value) is dict:
        for item in value.values():source_uuid_namespace(item,found,depth+1)
    elif type(value) is list:
        for item in value:source_uuid_namespace(item,found,depth+1)
    elif type(value) is str and re.fullmatch('[0-9a-f]{8}-(?:[0-9a-f]{4}-){3}[0-9a-f]{12}',value):found.add(value)
    return found


def closed_module(relative):
    """Every transitive runpy request compiles prehashed admitted bytes only."""
    import builtins
    need(relative in CLOSED_MODULE_BYTES,'CONTROL_HELPER_NOT_PREHASHED')
    module=ModuleType('_restricted_closed_'+relative.replace('/','_').replace('.','_'));module.__file__=str(ROOT/relative)
    shim=ModuleType('runpy')
    def run_path(path):
        target=Path(path).resolve()
        need(target.is_relative_to(ROOT.resolve()),'CONTROL_HELPER_PATH_ESCAPE')
        return closed_module(target.relative_to(ROOT.resolve()).as_posix())
    shim.run_path=run_path
    original_import=builtins.__import__
    def limited_import(name,*args,**kwargs):return shim if name=='runpy' else original_import(name,*args,**kwargs)
    module.__dict__['__builtins__']={**vars(builtins),'__import__':limited_import}
    stream=io.StringIO()
    try:
        with contextlib.redirect_stdout(stream),contextlib.redirect_stderr(stream):exec(compile(CLOSED_MODULE_BYTES[relative],module.__file__,'exec'),module.__dict__)
    except Stop:raise
    except BaseException:raise Stop('CONTROL_COORDINATOR_IMPORT_REFUSED') from None
    need(not stream.getvalue(),'CONTROL_COORDINATOR_IMPORT_OUTPUT')
    return module.__dict__


def verify_controls():
    cfg=MANIFEST['restricted_controls']
    need(type(cfg) is dict and set(cfg)=={'scope','order','components'},'RESTRICTED_CONTROL_FIELDS')
    exact(cfg['scope'],'FINITE_REVIEWED_SQL_PACKETS_NO_OWNER_ADAPTER_OR_WORKER_LOOP','RESTRICTED_CONTROL_SCOPE')
    order=cfg['order']
    need(type(order) is list and order and len(order)==len(set(order)) and [name for name in CONTROL_ORDER if name in order]==order,'CONTROL_SELECTED_ORDER')
    need(type(cfg['components']) is dict and set(cfg['components'])==set(CONTROL_ORDER),'CONTROL_COMPONENT_SET')
    # Every selected source/module/fixture graph is admitted before ANY
    # coordinator code. Unselected unfinished packets do not enter this run.
    packets={name:control_packet_graph(name,cfg['components'][name]) for name in order}
    namespaces={name:source_uuid_namespace(packets[name]) for name in order}
    for index,one in enumerate(order):
        for two in order[index+1:]:need(not namespaces[one]&namespaces[two],'CONTROL_FIXTURE_NAMESPACES_OVERLAP')
    for name in order:need(cfg['components'][name]['source_review']['decision']=='APPROVED_SOURCE','CONTROL_SOURCE_REVIEW_PENDING')
    # Include bootstrap/migration transitive runpy helpers. There is no disk
    # reread or helper-loading fallback after the bytes have been admitted.
    closure=set(path for name in order for path in CONTROL_SPECS[name]['modules'] if path.endswith('.py'))
    closure.update(('scripts/sql-review/run-first-rehearsal.py','scripts/sql-review/run-local-f0-migrate.py','scripts/sql-review/rehearsal-diagnostics.py','scripts/prepare-local-f0-history.py'))
    CLOSED_MODULE_BYTES.update({path:read_pinned(path) for path in sorted(closure)})
    modules={}
    for name in order:
        spec=CONTROL_SPECS[name]
        module=closed_module(spec['coordinator'])
        need(callable(module.get('main')),'CONTROL_COORDINATOR_ENTRYPOINT')
        declared=module.get('SOURCE_MODULES',module.get('MODULES'))
        exact(tuple(declared),spec['modules'],'CONTROL_TRANSITIVE_MODULE_CLOSURE')
        modules[name]=module
    CONTROLS.update(order=order,packets=packets,modules=modules)
    report['control_preparation']={'status':'SOURCE_REPRODUCED_NOT_RUN','selected_order':order,'fixture_uuid_namespace_sizes':{name:len(namespaces[name]) for name in order},'selected_fixture_namespaces_disjoint':True,'all_selected_graphs_in_committed_full_packet':True}

def control_result_summary(component,path,exit_code,require_complete=True):

    if path.is_file():

    else:value=None
